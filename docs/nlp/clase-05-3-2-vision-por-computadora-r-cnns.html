def plot(X, Y=None, xlabel=None, ylabel=None, legend=None, xlim=None,
         ylim=None, xscale='linear', yscale='linear',
         fmts=('-', 'm--', 'g-.', 'r:'), figsize=(3.5, 2.5), axes=None):
    """Plot data points.

    Defined in :numref:`sec_calculus`"""
    if legend is None:
        legend = []

    set_figsize(figsize)
    axes = axes if axes else plt.gca()

    # Return True if `X` (tensor or list) has 1 axis
    def has_one_axis(X):
        return (hasattr(X, "ndim") and X.ndim == 1 or isinstance(X, list)
                and not hasattr(X[0], "__len__"))

    if has_one_axis(X):
        X = [X]
    if Y is None:
        X, Y = [[]] * len(X), X
    elif has_one_axis(Y):
        Y = [Y]
    if len(X) != len(Y):
        X = X * len(Y)
    axes.cla()
    for x, y, fmt in zip(X, Y, fmts):
        if len(x):
            axes.plot(x, y, fmt)
        else:
            axes.plot(y, fmt)
    set_axes(axes, xlabel, ylabel, xlim, ylim, xscale, yscale, legend)

def show_images(imgs, num_rows, num_cols, titles=None, scale=1.5):
    """Plot a list of images.

    Defined in :numref:`sec_fashion_mnist`"""
    figsize = (num_cols * scale, num_rows * scale)
    _, axes = plt.subplots(num_rows, num_cols, figsize=figsize)
    axes = axes.flatten()

        ax.axes.get_xaxis().set_visible(False)
        ax.axes.get_yaxis().set_visible(False)
        if titles:
            ax.set_title(titles[i])
    return axes

class Accumulator:
    """For accumulating sums over `n` variables."""
    def __init__(self, n):
        """Defined in :numref:`sec_softmax_scratch`"""
        self.data = [0.0] * n

    def add(self, *args):
        self.data = [a + float(b) for a, b in zip(self.data, args)]

    def reset(self):
        self.data = [0.0] * len(self.data)

    def __getitem__(self, idx):
        return self.data[idx]

def get_dataloader_workers():

    return 4


def accuracy(y_hat, y):
    """Compute the number of correct predictions.

    Defined in :numref:`sec_softmax_scratch`"""
    if len(y_hat.shape) > 1 and y_hat.shape[1] > 1:
        y_hat = argmax(y_hat, axis=1)
    cmp = astype(y_hat, y.dtype) == y
    return float(reduce_sum(astype(cmp, y.dtype)))


def evaluate_accuracy(net, data_iter):
    """Compute the accuracy for a model on a dataset.

    Defined in :numref:`sec_softmax_scratch`"""
    if isinstance(net, torch.nn.Module):
        net.eval()  # Set the model to evaluation mode
    metric = Accumulator(2)  # No. of correct predictions, no. of predictions

    with torch.no_grad():
        for X, y in data_iter:
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def evaluate_loss(net, data_iter, loss):
    """Evaluate the loss of a model on the given dataset.

    Defined in :numref:`sec_model_selection`"""
    metric = Accumulator(2)  # Sum of losses, no. of examples
    for X, y in data_iter:
        out = net(X)
        y = reshape(y, out.shape)
        l = loss(out, y)
        metric.add(reduce_sum(l), size(l))
    return metric[0] / metric[1]

def try_gpu(i=0):
    """Return gpu(i) if exists, otherwise return cpu().

    Defined in :numref:`sec_use_gpu`"""
    if torch.cuda.device_count() >= i + 1:
        return torch.device(f'cuda:{i}')
    return torch.device('cpu')

def try_all_gpus():
    """Return all available GPUs, or [cpu(),] if no GPU exists.

    Defined in :numref:`sec_use_gpu`"""
    devices = [torch.device(f'cuda:{i}')
             for i in range(torch.cuda.device_count())]
    return devices if devices else [torch.device('cpu')]

def evaluate_accuracy_gpu(net, data_iter, device=None):
    """Compute the accuracy for a model on a dataset using a GPU.

    Defined in :numref:`sec_lenet`"""
    if isinstance(net, nn.Module):
        net.eval()  # Set the model to evaluation mode
        if not device:
            device = next(iter(net.parameters())).device
    # No. of correct predictions, no. of predictions
    metric = Accumulator(2)

    with torch.no_grad():
        for X, y in data_iter:
            if isinstance(X, list):
                # Required for BERT Fine-tuning (to be covered later)
                X = [x.to(device) for x in X]
            else:
                X = X.to(device)
            y = y.to(device)
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def train_ch6(net, train_iter, test_iter, num_epochs, lr, device):
    """Train a model with a GPU (defined in Chapter 6).

    Defined in :numref:`sec_lenet`"""
    def init_weights(m):
        if type(m) == nn.Linear or type(m) == nn.Conv2d:
            nn.init.xavier_uniform_(m.weight)
    net.apply(init_weights)
    print('training on', device)
    net.to(device)
    optimizer = torch.optim.SGD(net.parameters(), lr=lr)
    loss = nn.CrossEntropyLoss()
    animator = Animator(xlabel='epoch', xlim=[1, num_epochs],
                            legend=['train loss', 'train acc', 'test acc'])
    timer, num_batches = Timer(), len(train_iter)
    for epoch in range(num_epochs):
        # Sum of training loss, sum of training accuracy, no. of examples
        metric = Accumulator(3)
        net.train()
        for i, (X, y) in enumerate(train_iter):
            timer.start()
            optimizer.zero_grad()
            X, y = X.to(device), y.to(device)
            y_hat = net(X)
            l = loss(y_hat, y)
            l.backward()
            optimizer.step()
            with torch.no_grad():
                metric.add(l * X.shape[0], accuracy(y_hat, y), X.shape[0])
            timer.stop()
            train_l = metric[0] / metric[2]
            train_acc = metric[1] / metric[2]
            if (i + 1) % (num_batches // 5) == 0 or i == num_batches - 1:
                animator.add(epoch + (i + 1) / num_batches,
                             (train_l, train_acc, None))
        test_acc = evaluate_accuracy_gpu(net, test_iter)
        animator.add(epoch + 1, (None, None, test_acc))
    print(f'loss {train_l:.3f}, train acc {train_acc:.3f}, '
          f'test acc {test_acc:.3f}')
    print(f'{metric[2] * num_epochs / timer.sum():.1f} examples/sec '
          f'on {str(device)}')

class Animator:
    """For plotting data in animation."""
    def __init__(self, xlabel=None, ylabel=None, legend=None, xlim=None,
                 ylim=None, xscale='linear', yscale='linear',
                 fmts=('-', 'm--', 'g-.', 'r:'), nrows=1, ncols=1,
                 figsize=(3.5, 2.5)):
        """Defined in :numref:`sec_softmax_scratch`"""
        # Incrementally plot multiple lines
        if legend is None:
            legend = []
        use_svg_display()
        self.fig, self.axes = plt.subplots(nrows, ncols, figsize=figsize)
        if nrows * ncols == 1:
            self.axes = [self.axes, ]
        # Use a lambda function to capture arguments
        self.config_axes = lambda: set_axes(
            self.axes[0], xlabel, ylabel, xlim, ylim, xscale, yscale, legend)
        self.X, self.Y, self.fmts = None, None, fmts

    def add(self, x, y):
        # Add multiple data points into the figure
        if not hasattr(y, "__len__"):
            y = [y]
        n = len(y)
        if not hasattr(x, "__len__"):
            x = [x] * n
        if not self.X:
            self.X = [[] for _ in range(n)]
        if not self.Y:
            self.Y = [[] for _ in range(n)]
        for i, (a, b) in enumerate(zip(x, y)):
            if a is not None and b is not None:
                self.X[i].append(a)
                self.Y[i].append(b)
        self.axes[0].cla()
        for x, y, fmt in zip(self.X, self.Y, self.fmts):
            self.axes[0].plot(x, y, fmt)
        self.config_axes()
        display.display(self.fig)
        display.clear_output(wait=True)

class Timer:
    """Record multiple running times."""
    def __init__(self):
        """Defined in :numref:`subsec_linear_model`"""
        self.times = []
        self.start()

    def start(self):
        """Start the timer."""
        self.tik = time.time()


    def stop(self):
        """Stop the timer and record the time in a list."""
        self.times.append(time.time() - self.tik)
        return self.times[-1]


    def avg(self):
        """Return the average time."""
        return sum(self.times) / len(self.times)


    def sum(self):
        """Return the sum of time."""
        return sum(self.times)


    def cumsum(self):
        """Return the accumulated time."""
        return np.array(self.times).cumsum().tolist()

DATA_URL = 'http://d2l-data.s3-accelerate.amazonaws.com/'

def download(name, cache_dir=os.path.join('..', 'data')):
    """Download a file inserted into DATA_HUB, return the local filename.

    Defined in :numref:`sec_kaggle_house`"""
    assert name in DATA_HUB, f"{name} does not exist in {DATA_HUB}."

    url, sha1_hash      = DATA_HUB[name]
    os.makedirs(cache_dir, exist_ok=True)

    fname               = os.path.join(cache_dir, url.split('/')[-1])
    if os.path.exists(fname):
        sha1            = hashlib.sha1()
        with open(fname, 'rb') as f:
            while True:
                data    = f.read(1048576)
                if not data:
                    break
                sha1.update(data)

        if sha1.hexdigest() == sha1_hash:
            return fname  # Hit cache

    print(f'Downloading {fname} from {url}...')

    r                   = requests.get(url, stream=True, verify=True)

    with open(fname, 'wb') as f:
        f.write(r.content)
    return fname


def download_extract(name, folder=None):
    """Download and extract a zip/tar file.

    Defined in :numref:`sec_kaggle_house`"""

    fname           = download(name)
    base_dir        = os.path.dirname(fname)
    data_dir, ext   = os.path.splitext(fname)

    if ext == '.zip':
        fp          = zipfile.ZipFile(fname, 'r')
    elif ext in ('.tar', '.gz'):
        fp          = tarfile.open(fname, 'r')
    else:
        assert False, 'Only zip/tar files can be extracted.'

    fp.extractall(base_dir)
    
    return os.path.join(base_dir, folder) if folder else data_dir


def download_all():
    """Download all files in the DATA_HUB.

    Defined in :numref:`sec_kaggle_house`"""
    for name in DATA_HUB:
        download(name)

1.8 CNN Basadas en Regiones

En esta sección, presentaremos la R-CNN \cite{girshick2014rich} y su serie de mejoras: R-CNN rápida (\cite{girshick2015fast}), R-CNN más rápida (\cite{ren2015faster}), y la R-CNN máscara (\cite{he2017mask}).

1.8.1 R-CNN

R-CNN original obtiene propuestas de región con selective search y procesa cada región con una CNN. Los anchors aparecen en el RPN de Faster R-CNN; no son sinónimo de las propuestas originales.

Referencia: Ren et al. · Faster R-CNN

Modelo R-CNN.

Específicamente, la R-CNN consiste en los siguientes cuatro pasos:

  1. Calcular la búsqueda selectiva para extraer múltiples propuestas de región de alta calidad en la imagen de entrada. Estas regiones propuestas generalmente se seleccionan en múltiples escalas con diferentes dimensiones y tamaños. Cada propuesta de región se etiquetará con una clase y un cuadro delimitador de la gran verdad.

  2. Elegir una CNN preentrenada y modificar la capa de salida. Luego, se redimensione cada propuesta de región para el tamaño de entrada requerido por la red, y generar las características extraídas para la propuesta de región a través de la propagación hacia adelante.

  3. Tomar las características extraídas y clase etiqueta de cada propuesta de región como una muestra. Luego se entrenan múltiples máquina vectoriales de soporte (support vector machine - SVM) para clasificar objetos, donde cada svm determinada individualmente si la muestra contiene o no una clase específica.

  4. Tomar las características extraídas y el cuadro delimitador etiquetado para cada propuesta de región como muestra. Luego se entra un modelo de regresión lineal para predecir el cuadro delimitador de la gran verdad.

Aunque el modelo R-CNN usa una CNN preentrenada para extraer de manera efectiva las características de la imagen, es lento. Ya que supone, por ej., seleccionar miles de propuestas de región de una sola imagen de entrada: esto requiere miles de propagaciones hacia adelante de la CNN para realizar la detección de objetos. Este coste computacional hace que el uso generalizado de las R-CNN en el mundo real sea inviable.

1.8.1.1 Regresión de Cuadro Delimitador

Para una propuesta con centro y tamaño (px,py,pw,ph), un regresor predice offsets d. La caja ajustada se decodifica con g^x=px+pwdx, g^y=py+phdy, g^w=pwedw y g^h=phedh. Anchos y altos de la propuesta deben ser positivos. Los offsets provienen de características de la región, aunque se escriban de forma abreviada como d(p).

Referencia: Ren et al. · Faster R-CNN

Ilustración de la transformación entre la predicción y los cuadros delimitadores de la gran verdad. Imagen tomada de \cite{weng2017detection4}.

Para una caja real g y una propuesta p con tamaños positivos, los objetivos son tx=(gx−px)/pw, ty=(gy−py)/ph, tw=log(gw/pw) y th=log(gh/ph). Son valores reales sin un intervalo fijo de salida. R-CNN original usa regresores ajustados con pérdida cuadrática regularizada; Fast R-CNN integra una cabeza de regresión con smooth L1. No son el mismo entrenamiento.

Referencia: Ren et al. · Faster R-CNN

1.8.2 R-CNN Rápida (Fast R-CNN)

El principal cuello de botella de una R-CNN radica en la propagación hacia adelante de la CNN para cada propuesta de región, sin compartir el cómputo. Dado que estas regiones generalmente tienen superposiciones, las extracciones de características independientes conducen a muchos cálculos repetidos.

Una de las principales mejoras de la R-CNN rápida a partir de la R-CNN es que la propagación hacia adelante de la CNN solo se realiza en la imagen completa.

Modelo de la R-CNN Rápida

Fast R-CNN calcula un mapa convolucional de la imagen y aplica RoI pooling a n propuestas. Cada región se transforma a un tensor C×H_R×W_R; el batch de regiones tiene forma n×C×H_R×W_R. La cabeza produce probabilidades para K clases de objeto más fondo y ajustes de caja. La regresión original es específica por clase (4K valores para las clases de objeto); una cabeza de cuatro valores independiente de clase es una variante, no la misma parametrización.

Referencia: Ren et al. · Faster R-CNN

1.8.2.1 Función de Pérdida

Para una región con etiqueta u, la pérdida multitarea de Fast R-CNN es L=−logpu+λ1[u≥1]∑r∈{x,y,w,h}smoothL1(tru−vr). Aquí v es el objetivo de regresión codificado relativo a la propuesta, no las coordenadas brutas de la caja real. La rama de caja se omite para fondo. En la definición original, smooth_L1(z)=0.5z² si |z|<1 y |z|−0.5 en otro caso; λ controla su peso.

Referencia: Ren et al. · Faster R-CNN

Gráfica de la pérdida uniforme de L1, y=L1smooth(x). Imagen tomada de \cite{weng2017detection1}.

1.8.3 R-CNN más Rápida (Faster R-CNN)

Para ser más precisos en la detección de objetos, el modelo de R-CNN rápida usualemente tiene que generar una cantidad significativa de propuestas de regiones en la búsqueda selectiva. Para reducir las propuestas de regiones sin pérdida de la precisión, la R-CNN más rápida reemplaza la búsqueda selectiva con una red de propuesta de región.

Modelo de la R-CNN más rápida.

Este modelo solo cambia el método de propuesta de región de búsqueda selectiva a una red de propuesta de región. El resto del modelo permanece sin cambios. El funcionamiento de la red de propuestas de región se describe a continuación:

  • Se usa una capa convolucional de 3×3 con padding 1 para transformar la salida de la CNN en una nueva salida con c canales. De esta manera, cada unidad a lo largo de las dimensiones espaciales de los mapas de características extraídos de la CNN obtinen a nuevo vector de características de longitud c.

  • Centrado en cada píxel de los mapas de características, se generan múltiples cajas ancla de diferentes escalas y relaciones de aspectos y se los etiquetan.

  • Usando el vector de característica de longitud c en el centro de cada caja ancla, se predice la clase binaria (fondo o objetos) y el cuadro delimitador para esta caja ancla.

  • Se considera los cuadros delimitador predichos cuyas clases predichas son objetos. Luego, se eliminan los resultados superpuestos utilizando la supresión no máxima (non-maximum suppression - nms). Los cuadros delimitadores restantes para los objetos son las propuestas de regiones requeridas para la capa de pooling de la región de interés.

Cabe destacar que, como parte del modelo de la R-CNN más rápida, la red de propuesta de región es conjuntamente entrenada con el resto del modelo. Es decir, la función objetivo de las R-CNN más rápida incluye no solo la clase y las predicciones de los cuadros delimitadores en la detección de objetos, sino que también la clase binaria y las predicción del cuadro delimitador de las cajas ancla en la red de región propuesta. Como resultado, la red de propuestas de regiones aprende a generar propuestas de regiones de alta calidad para mantener la precisión en la detección de objetos con un número reducido de propuestas de regiones que se aprende de los datos.

1.8.3.1 Flujo de Trabajo del Modelo

El RPN de Faster R-CNN clasifica anchors y predice ajustes de cajas. En su etiquetado original, es positivo el anchor con mayor IoU para una caja real o con IoU superior a 0.7; por debajo de 0.3 es negativo si no fue asignado positivo. Los demás se ignoran. El artículo propuso entrenamiento alternado; las implementaciones conjuntas posteriores son otra elección.

Referencia: Ren et al. · Faster R-CNN

1.8.3.2 Función de Pérdida

La R-CNN más rápida está optimizada para una función de pérdida multitarea, similar a la R-CNN rápida. Sean:

  • pi, la probabilidad predicha de que el ancla i sea un objeto.

  • pi*, etiqueta de la gran verdad (binaria) si el ancla i es un objeto.

  • ti, coordenadas parametrizadas predichas (cuatro).

  • ti*, coordenadas de la gran verdad.

  • Ncls, término de normalización, establece un tamaño de minilote (256) según el artículo original.

  • Nbox, término de normalización, establece el número de ubicaciones de anclas (2400) en el artículo original.

  • λ, parámetro de balanceo, establecido en ~10 en el artículo original (de modo que los términos Ncls y Nbox tengan aproximadamente la misma ponderación).

La función de pérdida multitarea combina las pérdidas de las clasificaciones y la regresión de los cuadros delimitadores:

ℒ&=ℒcls+ℒboxℒ({pi},{ti})&=1Ncls∑iℒcls(pi,pi*)+λNbox∑ipi*·L1smooth(ti−ti*)

donde Ncls es la función de pérdida sobre dos clases, ya que podemos traducir fácilmente una clasificación de varias clases en una clasificación binaria prediciendo que una muestras es un objeto de destinto o no. L1smooth es la pérdida suave de L1.

ℒcls(pi,pi*)=−pi*logpi−(1−pi*)log(1−pi).

1.8.4 Máscara R-CNN (Mask R-CNN)

En el dataset de entrenamiento, si las posiciones de los objetos a nivel de píxel también se etiquetan en las imágenes, la máscara R-CNN puede aprovechar de manera efectiva dichas etiquetas detalladas para mejorar aún más la precisión de la detección de objetos.

Modelo de la máscara R-CNN

Mask R-CNN añade una rama de máscaras a Faster R-CNN y utiliza RoIAlign para extraer características alineadas por región. La interpolación bilineal permite evaluar posiciones fraccionarias sin la cuantización de límites de RoI pooling. Las cabezas predicen clase, ajustes de caja y una máscara por región.

Referencia: He et al. · Mask R-CNN

RoIAlign

RoIAlign evita redondear los límites y las coordenadas de muestreo de cada región. Usa interpolación bilineal de los mapas de características y agrega muestras dentro de cada bin de salida. Es especialmente relevante para preservar alineación espacial en segmentación.

Referencia: He et al. · Mask R-CNN

Predicciones de la máscara R-CNN sobre el dataset de prueba COCO. Fig. tomada de \cite{weng2017detection1}.

RoIAlign evita la cuantización de coordenadas usada por RoI pooling; no se trata de hashing. La escala espacial convierte coordenadas de imagen a coordenadas del mapa y la interpolación bilineal evalúa los puntos fraccionarios. Los detalles de alineación deben coincidir con la implementación usada.

Referencia: He et al. · Mask R-CNN

Una región de interés se mapea con precisión desde la imagen original al mapa de características sin redondear a números enteros. Fig. tomada de \cite{weng2017detection1}.

1.8.4.2 Función de Pérdida

La función de pérdida multitarea de la Máscara R-CNN combina la pérdida de la clasificación, la ubicación y la máscara de segmentación: ℒ=ℒcls+ℒbox+ℒmask donde ℒcls y ℒbox son las mismas que en la R-CNN más rápida.

Las ramas de la máscara agenera una máscara de dimensión m×m para cada RoI y clase; K clases en total. Por lo tanto, el total de salidas es de tamaño K·m2. Debido a que el modelo está tratando de aprender una máscara para cada clase, no hay competencia entre las clases para la generación de máscaras.

ℒmask se define como la pérdida de entropía cruzada binaria promedio, quer solo incluye la k−ésima máscara si la región está asociada con la clase de la gran verdad k.

ℒmask=−1m2∑1≤i,j≤m[yijlogy^ijk+(1−yij)log(1−y^ijk)],

donde yij es la etiqueta de la celda (i,j) en la máscara de verdad de la región de tamaño m×m; y^ijk es el valor predicho de la misma celda en la máscara aprendida para la clase de la gran verdad k.

1.9 Sólo se mira una vez (You Only Look Once - YOLO)

El modelo YOLO \cite{redmon2016you} es el primer intento de contruir un detector de objetos rápido en tiempo real. Debido a que YOLO no se somete al paso de propuesta de región y solo predice sobre un número limitado de cuadros delimitadores, puede hacer inferencias mucho más rápido.

1.9.1 Flujo de Trabajo

A continuación, se describe el flujo de trabajo para YOLO:

  1. Preentrenar una red CNN en la tarea de clasificación de imágenes.

  2. Dividir una imagen en S×S celdas. Si el centro de un objeto cae en una celda, esa celda es "responsable" de detectar la existencia de ese objeto. Cada celda predice: (a) la ubicación de los cuadros delimitadores B; (b) una puntuación de confianza y (c) una probabilidad condicionada de la clase de objeto en la existencia de un objeto en el cuadro delimitador.

*   **Las coordenadas** del cuadro delimitador son definidas por un túpla de 4 valores, **(coordenada x centro, coordenada y centro, ancho, alto), es decir, (x,y,w,h), donde x e y están configurados para desplazar la ubicación de una celda. Además, x,y,w y h son normalizados por el ancho y el alto de la imagen, y por lo tanto, están en el rango de (0,1].**

*   **Una puntuación de confianza** inde la **probabilidad de que la celda contenga un objeto: Pr(contener un objeto) x IoU(predicción, gran verdad); donde Pr es la probabilidad y IoU es la intersección sobre la unión.**

*   **Si la celda contiene un objeto, esta predice una probabilidad de que ese objeto pertenezca a cada clase Ci,i=1,…,K: Pr(el objeto pertenezca a la clase C_i | contiene un objeto). En esta etapa, el modelo solo predice un conjunto de probabilidades de clase por celda, independientemente del número de cuadros delimitadores, B.**  

*   **En total, una imagen contiene S×S×B cuadros delimitadores, cada cuadro se corresponde a 4 predicciones de ubicación, 1 puntaje de confianza y K probabilidades condicionales para la clasificación de objetos. Los valores de predicción para una imagen son S×S×(5B+K), que es la forma de tensor de la capa convolucional final del modelo.**
  1. La capa final de la CNN preentrenada es modificada para generar un tensor de predicción de tamaño S×S×(5B+K).

Flujo de trabajo del modelo YOLO. Fig. tomada de \cite{weng2017detection1}.

1.9.2 Aquitectura de la red

La base del modelo es similar al modelo GoogLeNet con el módulo Inception reemplazado por capas convolucionales 1×1 y 3×3. La predicción final es de dimensión S×S×(5B+K) se produce por dos capas completamente contectada en el todo el mapa de características convolucional.

La arquitectura tiene 24 capas convolucionales seguidas de dos capas completamente conectadas.

Arquitectura de YOLO. Fig. tomada de \cite{weng2017detection1}.

1.9.3 Función de Pérdida

YOLOv1 divide la imagen en S×S celdas. Predice B cajas por celda y C probabilidades de clase compartidas por sus cajas. La pérdida suma regresión de centro y raíz de tamaño de la caja responsable, error de confianza con distintos pesos para objeto/fondo y error de clase en celdas con objeto. Los índices de celda recorren 0,…,S²−1 y los de caja 0,…,B−1. Es un objetivo diferente del utilizado por YOLOv3; no se deben intercambiar sus parametrizaciones.

Referencia: Redmon y Farhadi · YOLO9000

En una ubicación, en la celda i, el modelo propone B candidatos de cuadros delimitadores y el que tiene más superposición con respecto a la gran verdad es el predictor "responsable". Fig. tomada de \cite{weng2017detection1}.

La función de pérdida solo penaliza el error de clasificación si un objeto está presente en la celda de la grilla, 1iobj=1. También solo penaliza el error de coordenadas del cuadro delimitador si ese predictor es "responsable" del cuadro de la gran verdad, 1ijobj=1.

Como detector de objetos de una etapa, YOLO es considerablemente rápido, pero no es bueno para reconocer objetos de forma irregular o un grupo de objetos pequeños debido a un número limitado de candidatos de cuadros delimitadores.

1.10 YOLOv2 / YOLO9000

El modelo YOLOv2 \cite{redmon2017yolo9000} es una versión mejorada de YOLO. YOLO9000 está construido sobre YOLOv2 pero entrenado con un dataset que combina el dataset de detección COCO y las 9000 clases principales de ImageNet.

1.10.1 YOLOv2

Recall se traduce como exhaustividad o sensibilidad: TP/(TP+FN). No es «memoria». Las mejoras y cifras de YOLOv2 pertenecen a los datasets y protocolos del artículo.

Referencia: Redmon y Farhadi · YOLO9000

Predicción de la ubicación del cuadro delimitador en el modelo YOLOv2. Fig. tomada de \cite{weng2017detection1}.

La capa passthrough de YOLOv2 reorganiza bloques espaciales 2×2 en canales y concatena esas características con un mapa de menor resolución. No es la suma identidad de un bloque residual ResNet.

Referencia: Redmon y Farhadi · YOLO9000

1.10.2 YOLO9000

YOLO9000 combina detección y clasificación mediante WordTree, una jerarquía de categorías con probabilidades condicionales entre nodos hermanos. El modelo del artículo cubre 9418 categorías; no debe confundirse con las 1000 clases de ImageNet usadas en otros experimentos.

Referencia: Redmon y Farhadi · YOLO9000

La jerarquía de WordTree combina etiquetas de COCO e ImageNet. Los nodos azules son etiquetas de COCO y los nodos rojos son etiquetas de ImageNet. Fig. tomada de \cite{weng2017detection1}.

Para predecir la probabilidad de un nodo de clase, podemos seguir el camino desde el nodo hasta la ráiz:

Pr("persian cat" | contain a "physical object") = \
    Pr("persian cat" | "cat")
    Pr("cat" | "animal")
    Pr("animal" | "pysical object")
    Pr(contain a "physical object") # puntuación de confianza

Cabe destacar que Pr(contain a "physical object") es la puntuación de confianza predecida por separado en la estructura de pipeline de cuadro delimitador. El camino de la predicción de la probabilidad condicional puede detenerse en cualquier paso, según las etiquetas disponibles.

1.11 RetinaNet

El modelo de RetinaNet \cite{lin2017focal} es un detector de objetos denso de una etapa. Dos bloques de construcción cruciales son: (1) la pirámide de imágenes destacada y (2) el uso de la pérdida focal.

La pirámide de imágenes se basa en la idea intuitiva de que mapas de características de grano fino grandes en niveles anteriores son buenos para captura objetos pequeños y los mapas de características de grano grueso pequeños pueden detectar bien objetos grandes.

Ejemplo de pirámide de imágenes del modelo SSD (previamente descrito), donde la detección ocurre en cada capa piramidal, apuntando a objetos de varios tamaños. Fig. tomada de \cite{weng2017detection1}.

1.11.1 Pérdida Focal (Focal Loss)

Para una decisión binaria de clase, pt=p si y=1 y pt=1−p si y=0. Focal loss es −αt(1−pt)γlogpt. En RetinaNet se usa clasificación sigmoide por clase y anchor; no es únicamente una pérdida de objectness.

Referencia: Lin et al. · Focal Loss

La pérdida focal se centra menos en ejemplos sencillos con un factor de (1−pt)γ. Fig. tomada de \cite{weng2017detection1}.

En la variante balanceada, αt=α para positivos y 1−α para negativos. El factor (1−pt)γ reduce el peso de ejemplos bien clasificados. Los valores γ=2 y α=0.25 son una configuración estudiada, no un óptimo universal.

Referencia: Lin et al. · Focal Loss

Gráfico de los pesos de la pérdida focal α(1−pt)γ en función de pt dados diferentes valores de α y γ. Fig. tomada de \cite{weng2017detection1}

1.11.2 Pirámide de Imágenes Destacadas

RetinaNet utiliza una pirámide de mapas de características FPN calculados por la red. Es diferente de ejecutar el modelo sobre una pirámide de imágenes redimensionadas.

Referencia: Lin et al. · Focal Loss

Pirámide de Imágenes destacadas. Fig. tomada de \cite{weng2017detection1}.

Dos vías conectan las capas convolucionales:

  1. Camino de abajo hacia arriba: es el cálculo hacia adelante normal.

  2. Camino de arriba hacia abajo: va en dirección inversa, agregando mapas de características semánticamente más fuertes de regreso a los niveles de pirámide anteriores de un tamaño más grande a través de conexiones laterales.

*    **Primero, las características de un nivel más alto se muestrean espacialmente más gruesas para que sean 2 veces más grandes. Para la mejora de la imagen, el artículo original utiliza el muestreo superio del vecino más cercano. Si bien existen muchos algoritmos de escalado de imágenes, como el uso de la deconvolución, la adopción de otro método de escalado de imágenes podría o no mejorar el rendimiento de RetinaNet.**

*    **El mapa de características más grande se somete a una capa de convolución de 1×1 para reducir la dimensión del canal.**  

*    **Finalmente, estos dos mapas de características se mezclan mediante la agregación de elementos inteligente.**

*    **Las conexiones laterales solo ocurren en la última capa en las etapas, denotado como {Ci}, y el proceso continúa hasta que se genera el mapa de características combinado más fino (más grande). La predicción se realiza a partir de cada mapa mezclado después de una capa de convolución de 3×3, {Pi}.**

Las comparaciones del artículo son resultados de sus experimentos y configuraciones. El ranking entre arquitecturas puede cambiar con el dataset, la métrica, la resolución y el presupuesto de cómputo.

Referencia: Lin et al. · Focal Loss

1.11.3 Arquitectura del Modelo

RetinaNet coloca nueve anchors por posición espacial: tres escalas por tres razones de aspecto. Los niveles P3–P7 tienen strides 8–128 y áreas base de 32²–512² píxeles. No son solo nueve cajas en todo un nivel.

Referencia: Lin et al. · Focal Loss

Arquitectura del modelo RetinaNet que utiliza una red troncal FPN sobre ResNet. Fig. tomada de \cite{weng2017detection1}.
