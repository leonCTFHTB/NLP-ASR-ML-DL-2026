def set_axes(axes, xlabel, ylabel, xlim, ylim, xscale, yscale, legend):
    """Set the axes for matplotlib.

    Defined in :numref:`sec_calculus`"""
    axes.set_xlabel(xlabel)

    axes.set_xscale(xscale)
    axes.set_yscale(yscale)

def plot(X, Y=None, xlabel=None, ylabel=None, legend=None, xlim=None,
         ylim=None, xscale='linear', yscale='linear',
         fmts=('-', 'm--', 'g-.', 'r:'), figsize=(3.5, 2.5), axes=None):
    """Plot data points.

    Defined in :numref:`sec_calculus`"""
    if legend is None:
        legend = []

    set_figsize(figsize)
    axes = axes if axes else plt.gca()

    # Return True if `X` (tensor or list) has 1 axis
    def has_one_axis(X):
        return (hasattr(X, "ndim") and X.ndim == 1 or isinstance(X, list)
                and not hasattr(X[0], "__len__"))

    if has_one_axis(X):
        X = [X]
    if Y is None:
        X, Y = [[]] * len(X), X
    elif has_one_axis(Y):
        Y = [Y]
    if len(X) != len(Y):
        X = X * len(Y)
    axes.cla()
    for x, y, fmt in zip(X, Y, fmts):
        if len(x):
            axes.plot(x, y, fmt)
        else:
            axes.plot(y, fmt)
    set_axes(axes, xlabel, ylabel, xlim, ylim, xscale, yscale, legend)

def show_images(imgs, num_rows, num_cols, titles=None, scale=1.5):
    """Plot a list of images.

    Defined in :numref:`sec_fashion_mnist`"""
    figsize = (num_cols * scale, num_rows * scale)
    _, axes = plt.subplots(num_rows, num_cols, figsize=figsize)
    axes = axes.flatten()
    for i, (ax, img) in enumerate(zip(axes, imgs)):
        if torch.is_tensor(img):
            # Tensor Image
            ax.imshow(img.numpy())
        else:
            # PIL Image
            ax.imshow(img)
        ax.axes.get_xaxis().set_visible(False)
        ax.axes.get_yaxis().set_visible(False)
        if titles:
            ax.set_title(titles[i])
    return axes

class Accumulator:
    """For accumulating sums over `n` variables."""
    def __init__(self, n):
        """Defined in :numref:`sec_softmax_scratch`"""
        self.data = [0.0] * n

    def add(self, *args):
        self.data = [a + float(b) for a, b in zip(self.data, args)]

    def reset(self):
        self.data = [0.0] * len(self.data)

    def __getitem__(self, idx):
        return self.data[idx]

def get_dataloader_workers():
    """Use 4 processes to read the data.

    return 4


def accuracy(y_hat, y):
    """Compute the number of correct predictions.

    Defined in :numref:`sec_softmax_scratch`"""
    if len(y_hat.shape) > 1 and y_hat.shape[1] > 1:
        y_hat = argmax(y_hat, axis=1)
    cmp = astype(y_hat, y.dtype) == y
    return float(reduce_sum(astype(cmp, y.dtype)))


def evaluate_accuracy(net, data_iter):
    """Compute the accuracy for a model on a dataset.

    Defined in :numref:`sec_softmax_scratch`"""
    if isinstance(net, torch.nn.Module):
        net.eval()  # Set the model to evaluation mode
    metric = Accumulator(2)  # No. of correct predictions, no. of predictions

    with torch.no_grad():
        for X, y in data_iter:
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def evaluate_loss(net, data_iter, loss):
    """Evaluate the loss of a model on the given dataset.

    Defined in :numref:`sec_model_selection`"""
    metric = Accumulator(2)  # Sum of losses, no. of examples
    for X, y in data_iter:
        out = net(X)
        y = reshape(y, out.shape)
        l = loss(out, y)
        metric.add(reduce_sum(l), size(l))
    return metric[0] / metric[1]

def try_gpu(i=0):
    """Return gpu(i) if exists, otherwise return cpu().

    Defined in :numref:`sec_use_gpu`"""
    if torch.cuda.device_count() >= i + 1:
        return torch.device(f'cuda:{i}')
    return torch.device('cpu')

def try_all_gpus():
    """Return all available GPUs, or [cpu(),] if no GPU exists.

    Defined in :numref:`sec_use_gpu`"""
    devices = [torch.device(f'cuda:{i}')
             for i in range(torch.cuda.device_count())]
    return devices if devices else [torch.device('cpu')]

def evaluate_accuracy_gpu(net, data_iter, device=None):
    """Compute the accuracy for a model on a dataset using a GPU.

    Defined in :numref:`sec_lenet`"""
    if isinstance(net, nn.Module):
        net.eval()  # Set the model to evaluation mode
        if not device:
            device = next(iter(net.parameters())).device
    # No. of correct predictions, no. of predictions
    metric = Accumulator(2)

    with torch.no_grad():
        for X, y in data_iter:
            if isinstance(X, list):
                # Required for BERT Fine-tuning (to be covered later)
                X = [x.to(device) for x in X]
            else:
                X = X.to(device)
            y = y.to(device)
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def train_ch6(net, train_iter, test_iter, num_epochs, lr, device):
    """Train a model with a GPU (defined in Chapter 6).

    Defined in :numref:`sec_lenet`"""
    def init_weights(m):
        if type(m) == nn.Linear or type(m) == nn.Conv2d:
            nn.init.xavier_uniform_(m.weight)
    net.apply(init_weights)
    print('training on', device)
    net.to(device)
    optimizer = torch.optim.SGD(net.parameters(), lr=lr)
    loss = nn.CrossEntropyLoss()
    animator = Animator(xlabel='epoch', xlim=[1, num_epochs],
                            legend=['train loss', 'train acc', 'test acc'])
    timer, num_batches = Timer(), len(train_iter)
    for epoch in range(num_epochs):
        # Sum of training loss, sum of training accuracy, no. of examples
        metric = Accumulator(3)
        net.train()
        for i, (X, y) in enumerate(train_iter):
            timer.start()
            optimizer.zero_grad()
            X, y = X.to(device), y.to(device)
            y_hat = net(X)
            l = loss(y_hat, y)
            l.backward()
            optimizer.step()
            with torch.no_grad():
                metric.add(l * X.shape[0], accuracy(y_hat, y), X.shape[0])
            timer.stop()
            train_l = metric[0] / metric[2]
            train_acc = metric[1] / metric[2]
            if (i + 1) % (num_batches // 5) == 0 or i == num_batches - 1:
                animator.add(epoch + (i + 1) / num_batches,
                             (train_l, train_acc, None))
        test_acc = evaluate_accuracy_gpu(net, test_iter)
        animator.add(epoch + 1, (None, None, test_acc))
    print(f'loss {train_l:.3f}, train acc {train_acc:.3f}, '
          f'test acc {test_acc:.3f}')
    print(f'{metric[2] * num_epochs / timer.sum():.1f} examples/sec '
          f'on {str(device)}')

class Animator:
    """For plotting data in animation."""
    def __init__(self, xlabel=None, ylabel=None, legend=None, xlim=None,
                 ylim=None, xscale='linear', yscale='linear',
                 fmts=('-', 'm--', 'g-.', 'r:'), nrows=1, ncols=1,
                 figsize=(3.5, 2.5)):
        """Defined in :numref:`sec_softmax_scratch`"""
        # Incrementally plot multiple lines
        if legend is None:
            legend = []
        use_svg_display()
        self.fig, self.axes = plt.subplots(nrows, ncols, figsize=figsize)
        if nrows * ncols == 1:
            self.axes = [self.axes, ]
        # Use a lambda function to capture arguments
        self.config_axes = lambda: set_axes(
            self.axes[0], xlabel, ylabel, xlim, ylim, xscale, yscale, legend)
        self.X, self.Y, self.fmts = None, None, fmts

    def add(self, x, y):
        # Add multiple data points into the figure
        if not hasattr(y, "__len__"):
            y = [y]
        n = len(y)
        if not hasattr(x, "__len__"):
            x = [x] * n
        if not self.X:
            self.X = [[] for _ in range(n)]
        if not self.Y:
            self.Y = [[] for _ in range(n)]
        for i, (a, b) in enumerate(zip(x, y)):
            if a is not None and b is not None:
                self.X[i].append(a)
                self.Y[i].append(b)
        self.axes[0].cla()
        for x, y, fmt in zip(self.X, self.Y, self.fmts):
            self.axes[0].plot(x, y, fmt)
        self.config_axes()
        display.display(self.fig)
        display.clear_output(wait=True)

class Timer:
    """Record multiple running times."""
    def __init__(self):
        """Defined in :numref:`subsec_linear_model`"""
        self.times = []
        self.start()

    def start(self):
        """Start the timer."""
        self.tik = time.time()


    def stop(self):
        """Stop the timer and record the time in a list."""
        self.times.append(time.time() - self.tik)
        return self.times[-1]


    def avg(self):
        """Return the average time."""
        return sum(self.times) / len(self.times)


    def sum(self):
        """Return the sum of time."""
        return sum(self.times)


    def cumsum(self):
        """Return the accumulated time."""
        return np.array(self.times).cumsum().tolist()

DATA_HUB = dict()
DATA_URL = 'http://d2l-data.s3-accelerate.amazonaws.com/'

def download(name, cache_dir=os.path.join('..', 'data')):
    """Download a file inserted into DATA_HUB, return the local filename.

    Defined in :numref:`sec_kaggle_house`"""
    assert name in DATA_HUB, f"{name} does not exist in {DATA_HUB}."

    url, sha1_hash      = DATA_HUB[name]
    os.makedirs(cache_dir, exist_ok=True)

    fname               = os.path.join(cache_dir, url.split('/')[-1])
    if os.path.exists(fname):
        sha1            = hashlib.sha1()
        with open(fname, 'rb') as f:
            while True:
                data    = f.read(1048576)
                if not data:
                    break
                sha1.update(data)

        if sha1.hexdigest() == sha1_hash:
            return fname  # Hit cache

    print(f'Downloading {fname} from {url}...')

    r                   = requests.get(url, stream=True, verify=True)

    with open(fname, 'wb') as f:
        f.write(r.content)
    return fname


def download_extract(name, folder=None):
    """Download and extract a zip/tar file.

    Defined in :numref:`sec_kaggle_house`"""

    fname           = download(name)
    base_dir        = os.path.dirname(fname)
    data_dir, ext   = os.path.splitext(fname)

    if ext == '.zip':
        fp          = zipfile.ZipFile(fname, 'r')
    elif ext in ('.tar', '.gz'):
        fp          = tarfile.open(fname, 'r')
    else:
        assert False, 'Only zip/tar files can be extracted.'

    fp.extractall(base_dir)
    
    return os.path.join(base_dir, folder) if folder else data_dir


def download_all():
    """Download all files in the DATA_HUB.

    Defined in :numref:`sec_kaggle_house`"""
    for name in DATA_HUB:
        download(name)

1.8 CNN Basadas en Regiones

En esta sección, presentaremos la R-CNN \cite{girshick2014rich} y su serie de mejoras: R-CNN rápida (\cite{girshick2015fast}), R-CNN más rápida (\cite{ren2015faster}), y la R-CNN máscara (\cite{he2017mask}).

1.8.1 R-CNN

La R-CNN primero extrae muchas propuestas de región (por ej., 2000) de la imagen de entrada (por ej., las cajas anclas se pueden considerar como propuestas de región), etiquetando sus clases y cuadros delimitadores (por ej., desplazamientos).

Entonces, una CNN es usada para calcular la propagación hacia adelante sobre cada propuesta de región para extraer sus características. A continuación, las características de cada propuesta de región son usadas para predicir la clases y el cuadro delimitador de dicha región.

Modelo R-CNN.

Específicamente, la R-CNN consiste en los siguientes cuatro pasos:

  1. Calcular la búsqueda selectiva para extraer múltiples propuestas de región de alta calidad en la imagen de entrada. Estas regiones propuestas generalmente se seleccionan en múltiples escalas con diferentes dimensiones y tamaños. Cada propuesta de región se etiquetará con una clase y un cuadro delimitador de la gran verdad.

  2. Elegir una CNN preentrenada y modificar la capa de salida. Luego, se redimensione cada propuesta de región para el tamaño de entrada requerido por la red, y generar las características extraídas para la propuesta de región a través de la propagación hacia adelante.

  3. Tomar las características extraídas y clase etiqueta de cada propuesta de región como una muestra. Luego se entrenan múltiples máquina vectoriales de soporte (support vector machine - SVM) para clasificar objetos, donde cada svm determinada individualmente si la muestra contiene o no una clase específica.

  4. Tomar las características extraídas y el cuadro delimitador etiquetado para cada propuesta de región como muestra. Luego se entra un modelo de regresión lineal para predecir el cuadro delimitador de la gran verdad.

Aunque el modelo R-CNN usa una CNN preentrenada para extraer de manera efectiva las características de la imagen, es lento. Ya que supone, por ej., seleccionar miles de propuestas de región de una sola imagen de entrada: esto requiere miles de propagaciones hacia adelante de la CNN para realizar la detección de objetos. Este coste computacional hace que el uso generalizado de las R-CNN en el mundo real sea inviable.

1.8.1.1 Regresión de Cuadro Delimitador

Dadas las coordenadas de un cuadro delimitador predicho 𝐩=(px,py,pw,ph) (coordenadas centrales, ancho, alto) y sus correspondientes coordenadas de caja de la gran verdad 𝐠=(gx,gy,gw,gh) el regresor está configurado para aprender la transformación invariante de escala entre dos centros y la transformación de escala logarítmica entre anchos y altos Todas las funciones de transformación toman 𝐩 como entrada.

\begin{aligned} \hat{g}_x &= p_w d_x(\mathbf{p}) + p_x \ \hat{g}_y &= p_h d_y(\mathbf{p}) + p_y \ \hat{g}_w &= p_w \exp({d_w(\mathbf{p})}) \ \hat{g}_h &= p_h \exp({d_h(\mathbf{p})}) \end{aligned}

Ilustración de la transformación entre la predicción y los cuadros delimitadores de la gran verdad. Imagen tomada de \cite{weng2017detection4}.

Un beneficio obvio de aplicar dicha transformación es que todas las funciones de corrección de los cuadros delimitadores, di(𝐩) donde i∈{x,y,w,h}, pueden tomar cualquier valor entre [−∞,+∞]. Los objetos para que aprendan son:

\begin{aligned} t_x &= (g_x - p_x) / p_w \ t_y &= (g_y - p_y) / p_h \ t_w &= \log(g_w/p_w) \ t_h &= \log(g_h/p_h). \end{aligned}

**Un modelo de regresión estándar puede resolver el problema al minimizar la pérdida de SSE con la regularización: **

\begin{aligned} \mathcal{L}\text{reg} = \sum{i \in {x, y, w, h}} (t_i - d_i(\mathbf{p}))^2 + \lambda |\mathbf{w}|^2. \end{aligned}

El término de regularización es crítico aquí, ya que el artículo orignal seleccionó el mejor λ mediante la validación cruzada (cross-validation). Por otra parte, no todos los cuadros delimitadores predichos tienen su correspondiente en la gran verdad. Por ej., si no hay superposición, no tiene sentido ejecutar la regresión bbox. Aquí, solo se mantiene un cuadro predicho con un cuadro de la gran verdad cercano con al menos 0.6 de IoU para entrenar el modelo de regresión de bbox.

1.8.2 R-CNN Rápida (Fast R-CNN)

El principal cuello de botella de una R-CNN radica en la propagación hacia adelante de la CNN para cada propuesta de región, sin compartir el cómputo. Dado que estas regiones generalmente tienen superposiciones, las extracciones de características independientes conducen a muchos cálculos repetidos.

Una de las principales mejoras de la R-CNN rápida a partir de la R-CNN es que la propagación hacia adelante de la CNN solo se realiza en la imagen completa.

Modelo de la R-CNN Rápida

Los principales cálculos de la R-CNN Rápida son los siguientes:

  1. Comparado con la R-CNN, en la R-CNN rápida, la entrada de la CNN para la extracción de características es la imagen de entrada, en lugar de las propuetas de regiones individuales. Además, esta CNN es entrenable. Dada una imagen de entrada, sea la dimensión de la salida de la CNN 1×c×h1×w1.

  2. Supongamos que la búsqueda selectiva genera n propuestas de regiones. Estas propuestas de regiones (de diferentes dimensiones) marcan regiones de interés (de diferentes dimensiones) en la salida de la CNN. Entonces, estas regiones de interés extraen aún más las características de la misma dimensión (por ej., se especifican la altura h2 y el ancho w2) para poder concatenarlas fácilmente. Para lograr esto, la R-CNN rápida introduce la capa de "pool de región de interés (Rol)": la salida de la CNN y las propuestas de región se ingresan en esta capa, lo que genera características concatenadas de dimensión c×c×h2×w2 que se extraen aún más para todas las propuestas de región.

  3. Usando una capa completamente conectada, se transforman las características concatenadas en una salida de dimensión n×d, donde d depende del diseño del modelo.

  4. Se predice la clase y el cuadro delimitador de cada una de las n propuestas de regiones. Más concretamente, en la predicción del cuadro delimitador y la clase, se transforma la salida de la capa completamente conectada en una salida de la forma n×q (q es el número de clases) y una salida de la forma n×4, respectivamente. La predicción de clase utiliza la regresión softmax.

La capa de pool de la región de interés propuesta en la R-CNN rápida es diferente de la capa de pool en la R-CNN. En la capa de pool, controlamos indirectamente la dimensión de salida especificando los tamaños de la ventana de pool, padding, y stride. Por otra parte, podemos especificar directamente la dimensión de la salida en la región de intrés de la capa de pool.

Por ej., sea la altura y el ancho de la salida para cada región como h2 y w2, respectivamente. Para cualquier ventana de región de interés de dimensión h×w, esta ventana se divide en una grilla de subventanas de h2×w2, donde la dimensión de cada subventana es aproximadamente (h/h2)×(w/w2). En la práctica, el alto y el ancho de cualquier subventana debería ser truncado hacia arriba, y el elemento más grande debería ser usado como la salida de la subventana. Por lo tanto, la capa de pool de la región de interés puede extraer características de la misma dimensión incluse cuando las regiones de interés tienen dimensiones diferentes.

Como un ejemplo, en la fig. de abajo, la región de interés de 3×3 es seleccionada en una entrada de 4×4. Para esta región de interés, usamos una capa de pool de región de interés de 2×2 para obtener una salida de 2×2. Hay que tener en cuenta que cada una de las cuatro subventanas divididas contienen los elementos 0,1,4 y 5 (5 es el máximo); 2 y 6 (6 es el máximo); 8 y 9 (9 es el máximo); y 10.

1.8.2.1 Función de Pérdida

El modelo es optimizado para una pérdida combinando dos tareas (clasificación y localización). Sean:

  • u, la etiqueta de clase verdadera, u∈0,1,…,K; por convención, la clase de fondo general tiene u=0.

  • p, la distribución de probabilidad discreta (por Rol) sobre K+1 clases: p=(p0,…,pK), calculada por una softmax sobre las K+1 salidas de una capa totalmente conectada.

  • v, el cuadro delimitador de verdad v=(vx,vy,vw,vh).

  • tu, la corrección del cuadro delimitador predicha, tu=(txu,tyu,twu,thu).

La función de pérdida resume el costo de la clasificación y la predicción del cuadro delimitador: ℒ=ℒcls+ℒbox. Para el "fondo" Rol, ℒbox es ignorado por el indicador de función 1[u≥1], definido como:

1[u>=1]={1if u≥10c.o.c.

Luego, la función de pérdidad global es:

ℒ(p,u,tu,v)&=ℒcls(p,u)+1[u≥1]ℒbox(tu,v)ℒcls(p,u)&=−logpuℒbox(tu,v)&=∑i∈{x,y,w,h}L1smooth(tiu−vi)

La pérdida del cuadro delimitador ℒbox debe medir la diferencia entre tiu y vi usando una función de pérdida robusta. En este caso, se adopta la pérdida uniforme de L1 (smooth L1 loss) y se afirma que es menos sensible a los valores atípicos.

L1smooth(x)={0.5x2if |x|<1|x|−0.5otherwise

1.8.3 R-CNN más Rápida (Faster R-CNN)

Para ser más precisos en la detección de objetos, el modelo de R-CNN rápida usualemente tiene que generar una cantidad significativa de propuestas de regiones en la búsqueda selectiva. Para reducir las propuestas de regiones sin pérdida de la precisión, la R-CNN más rápida reemplaza la búsqueda selectiva con una red de propuesta de región.

Modelo de la R-CNN más rápida.

Este modelo solo cambia el método de propuesta de región de búsqueda selectiva a una red de propuesta de región. El resto del modelo permanece sin cambios. El funcionamiento de la red de propuestas de región se describe a continuación:

  • Se usa una capa convolucional de 3×3 con padding 1 para transformar la salida de la CNN en una nueva salida con c canales. De esta manera, cada unidad a lo largo de las dimensiones espaciales de los mapas de características extraídos de la CNN obtinen a nuevo vector de características de longitud c.

  • Centrado en cada píxel de los mapas de características, se generan múltiples cajas ancla de diferentes escalas y relaciones de aspectos y se los etiquetan.

  • Usando el vector de característica de longitud c en el centro de cada caja ancla, se predice la clase binaria (fondo o objetos) y el cuadro delimitador para esta caja ancla.

  • Se considera los cuadros delimitador predichos cuyas clases predichas son objetos. Luego, se eliminan los resultados superpuestos utilizando la supresión no máxima (non-maximum suppression - nms). Los cuadros delimitadores restantes para los objetos son las propuestas de regiones requeridas para la capa de pooling de la región de interés.

Cabe destacar que, como parte del modelo de la R-CNN más rápida, la red de propuesta de región es conjuntamente entrenada con el resto del modelo. Es decir, la función objetivo de las R-CNN más rápida incluye no solo la clase y las predicciones de los cuadros delimitadores en la detección de objetos, sino que también la clase binaria y las predicción del cuadro delimitador de las cajas ancla en la red de región propuesta. Como resultado, la red de propuestas de regiones aprende a generar propuestas de regiones de alta calidad para mantener la precisión en la detección de objetos con un número reducido de propuestas de regiones que se aprende de los datos.

1.8.3.1 Flujo de Trabajo del Modelo

A continuación, se detalla el funcionamiento del flujo de trabajo del modelo de la R-CNN más rápida:

  1. Preentrenar una red CNN en la tarea de clasificación de imágenes.

  2. Ajustar la RPN (region proposal network), aplicando técnicas de transferencia de conocimiento (transfer learning, fine-tune) de extremo a extremo para la tarea de propuesta de región, que se inicializa mediante el clasificador de imágenes preentrenado. Las muestras positivas tienen un IoU (intersection-over-union) mayor a 0.7, mientras que las muestras negativas tienen un IoU<0.3.

*   **Deslizar una pequeña ventana espacial n×n sobre el mapa de características *conv* de toda la imagen.**

*   **En el centro de cada ventana de deslizamiento, se predice múltiples regiones de varias escalas y proporciones simultáneamente. Un ancla es una combinación de (centro de ventana deslizante, escala, proporciones).**

  1. Entrenar el modelo R-CNN más rápido, usando las propuestas generadas por el actual RPN.

  2. Utilizar la red R-CNN rápida para inicializar el entrenamiento de la RPN. MIentras se mantiene las capas convolucionales compartidas, solo se necesita ajustar las capas específicas de la RPN.

  3. Finalmente, se ajustan las capas únicas de la R-CNN rápida.

  4. Los pasos 4 y 5 pueden ser repetidos para entrenar la RPN y la R-CNN rápida.

1.8.3.2 Función de Pérdida

La R-CNN más rápida está optimizada para una función de pérdida multitarea, similar a la R-CNN rápida. Sean:

  • pi, la probabilidad predicha de que el ancla i sea un objeto.

  • pi*, etiqueta de la gran verdad (binaria) si el ancla i es un objeto.

  • ti, coordenadas parametrizadas predichas (cuatro).

  • ti*, coordenadas de la gran verdad.

  • Ncls, término de normalización, establece un tamaño de minilote (256) según el artículo original.

  • Nbox, término de normalización, establece el número de ubicaciones de anclas (2400) en el artículo original.

  • λ, parámetro de balanceo, establecido en ~10 en el artículo original (de modo que los términos Ncls y Nbox tengan aproximadamente la misma ponderación).

La función de pérdida multitarea combina las pérdidas de las clasificaciones y la regresión de los cuadros delimitadores:

ℒ&=ℒcls+ℒboxℒ({pi},{ti})&=1Ncls∑iℒcls(pi,pi*)+λNbox∑ipi*·L1smooth(ti−ti*)

donde Ncls es la función de pérdida sobre dos clases, ya que podemos traducir fácilmente una clasificación de varias clases en una clasificación binaria prediciendo que una muestras es un objeto de destinto o no. L1smooth es la pérdida suave de L1.

ℒcls(pi,pi*)=−pi*logpi−(1−pi*)log(1−pi).

1.8.4 Máscara R-CNN (Mask R-CNN)

En el dataset de entrenamiento, si las posiciones de los objetos a nivel de píxel también se etiquetan en las imágenes, la máscara R-CNN puede aprovechar de manera efectiva dichas etiquetas detalladas para mejorar aún más la precisión de la detección de objetos.

Modelo de la máscara R-CNN

Como se puede ver en la fig., la máscara R-CNN es una modificación basada en la R-CNN más rápida. Específicamente, la máscara R-CNN reemplaza la capa de pooling de la región de interés por la capa de alineamiento de la región de interés (Rol). Esta capa de región de alineamiento de región de interés usa una interpolación bilineal para preservar la información espacial en los mapas de características, lo cual es más adeacuado para la predicción a nivel de píxel.

La salida de esta capa contiene los mapas de características de la misma dimensión para todas las regiones de interés. Estos son usados para predecir no solo la clase y el cuadro delimitador para cada región de interés, sino que también la ubicación a nivel de píxel del objeto a través de una red completamente convolucional.

1.8.4.1 Rol Align

Debido a que la segmentación a nivel de píxel requiere una alineación mucho más detallada o final que los cuadros delimitadores, la máscara R-CNN mejora la capa de Rol (denominada "capa RolAlign") para que el Rol se pueda asignar mejor y con mayor precisión a las regiones de la imagen original.

Predicciones de la máscara R-CNN sobre el dataset de prueba COCO. Fig. tomada de \cite{weng2017detection1}.

La capa de RolAlign es diseñada para corregir la desalineación de la ubicación causada por la cuantificación en el pooling Rol. RolAlign elimina la cuantificación hash, por ej., usando x/16 en lugar de [x/16], para que las características extraídas puedan ser alineadas correctamente con los píxeles de entrada. La interpolación bilineal se utiliza para calcular los valores de ubicación de coma flotante en la entrada.

Una región de interés se mapea con precisión desde la imagen original al mapa de características sin redondear a números enteros. Fig. tomada de \cite{weng2017detection1}.

1.8.4.2 Función de Pérdida

La función de pérdida multitarea de la Máscara R-CNN combina la pérdida de la clasificación, la ubicación y la máscara de segmentación: ℒ=ℒcls+ℒbox+ℒmask donde ℒcls y ℒbox son las mismas que en la R-CNN más rápida.

Las ramas de la máscara agenera una máscara de dimensión m×m para cada RoI y clase; K clases en total. Por lo tanto, el total de salidas es de tamaño K·m2. Debido a que el modelo está tratando de aprender una máscara para cada clase, no hay competencia entre las clases para la generación de máscaras.

ℒmask se define como la pérdida de entropía cruzada binaria promedio, quer solo incluye la k−ésima máscara si la región está asociada con la clase de la gran verdad k.

ℒmask=−1m2∑1≤i,j≤m[yijlogy^ijk+(1−yij)log(1−y^ijk)],

donde yij es la etiqueta de la celda (i,j) en la máscara de verdad de la región de tamaño m×m; y^ijk es el valor predicho de la misma celda en la máscara aprendida para la clase de la gran verdad k.

1.9 Sólo se mira una vez (You Only Look Once - YOLO)

El modelo YOLO \cite{redmon2016you} es el primer intento de contruir un detector de objetos rápido en tiempo real. Debido a que YOLO no se somete al paso de propuesta de región y solo predice sobre un número limitado de cuadros delimitadores, puede hacer inferencias mucho más rápido.

1.9.1 Flujo de Trabajo

A continuación, se describe el flujo de trabajo para YOLO:

  1. Preentrenar una red CNN en la tarea de clasificación de imágenes.

  2. Dividir una imagen en S×S celdas. Si el centro de un objeto cae en una celda, esa celda es "responsable" de detectar la existencia de ese objeto. Cada celda predice: (a) la ubicación de los cuadros delimitadores B; (b) una puntuación de confianza y (c) una probabilidad condicionada de la clase de objeto en la existencia de un objeto en el cuadro delimitador.

*   **Las coordenadas** del cuadro delimitador son definidas por un túpla de 4 valores, **(coordenada x centro, coordenada y centro, ancho, alto), es decir, (x,y,w,h), donde x e y están configurados para desplazar la ubicación de una celda. Además, x,y,w y h son normalizados por el ancho y el alto de la imagen, y por lo tanto, están en el rango de (0,1].**

*   **Una puntuación de confianza** inde la **probabilidad de que la celda contenga un objeto: Pr(contener un objeto) x IoU(predicción, gran verdad); donde Pr es la probabilidad y IoU es la intersección sobre la unión.**

*   **Si la celda contiene un objeto, esta predice una probabilidad de que ese objeto pertenezca a cada clase Ci,i=1,…,K: Pr(el objeto pertenezca a la clase C_i | contiene un objeto). En esta etapa, el modelo solo predice un conjunto de probabilidades de clase por celda, independientemente del número de cuadros delimitadores, B.**  

*   **En total, una imagen contiene S×S×B cuadros delimitadores, cada cuadro se corresponde a 4 predicciones de ubicación, 1 puntaje de confianza y K probabilidades condicionales para la clasificación de objetos. Los valores de predicción para una imagen son S×S×(5B+K), que es la forma de tensor de la capa convolucional final del modelo.**
  1. La capa final de la CNN preentrenada es modificada para generar un tensor de predicción de tamaño S×S×(5B+K).

Flujo de trabajo del modelo YOLO. Fig. tomada de \cite{weng2017detection1}.

1.9.2 Aquitectura de la red

La base del modelo es similar al modelo GoogLeNet con el módulo Inception reemplazado por capas convolucionales 1×1 y 3×3. La predicción final es de dimensión S×S×(5B+K) se produce por dos capas completamente contectada en el todo el mapa de características convolucional.

La arquitectura tiene 24 capas convolucionales seguidas de dos capas completamente conectadas.

Arquitectura de YOLO. Fig. tomada de \cite{weng2017detection1}.

1.9.3 Función de Pérdida

La función de pérdida consiste en dos partes: (1) la pérdida de localización para la predicción del desplazamiento del cuadro delimitador y (2) la pérdida de la clasificación para las probabilidades de clase condicional.

Ambas partes son calculadas como la suma de los errores cuadráticos. Se utilizan dos parámetros de escala para controlar cuánto se quiere aumentar la pérdida de las predicciones de las coordenadas del cuadro delimitador (λcoord) y cuanto se quiere disminuir las predicciones de puntuación de pérdida de confianza para las cajas sin objetos (λnoobj). Es importante reducir la pérdida aportada por los cuadros de fondo, ya que la mayoría de los cuadros delimitadores no implican ninguna instancia. En el artículo original, se establecen: λcoord=5 y λnoobj=0.5.

Además, en el artículo original, la función de pérdida usa Ci en lugar de Cij como puntuación de confianza. Esta acotación está tomada de la siguiente observación: cada cuadro delimitador debe tener su propia puntuación de confianza.

A continuación, se muestra la función de pérdida:

\begin{aligned} \mathcal{L}\text{loc} &= \lambda\text{coord} \sum_{i=0}^{S^2} \sum_{j=0}^B \mathbb{1}{ij}^\text{obj} [(x_i - \hat{x}i)^2 + (y_i - \hat{y}i)^2 + (\sqrt{w_i} - \sqrt{\hat{w}i})^2 + (\sqrt{h_i} - \sqrt{\hat{h}i})^2 ] \ \mathcal{L}\text{cls} &= \sum{i=0}^{S^2} \sum{j=0}^B \big( \mathbb{1}{ij}^\text{obj} + \lambda\text{noobj} (1 - \mathbb{1}{ij}^\text{obj})\big) (C{ij} - \hat{C}{ij})^2 + \sum{i=0}^{S^2} \sum_{c \in \mathcal{C}} \mathbb{1}i^\text{obj} (p_i(c) - \hat{p}i(c))^2\ \mathcal{L} &= \mathcal{L}\text{loc} + \mathcal{L}\text{cls}, \end{aligned}

donde,

  • 1iobj, es una función indicadora de si la celda i contiene un objeto.

  • 1ijobj, indica si el cuadro delimitador j−ésimo de la celda i es "responsable" de la predicción del objeto. (ver fig. de abajo).

  • Cij, la puntuación de confianza de la celda i, Pr(contener un objeto) * IoU(pred, verdad).

  • C^ij, la puntuación de confianza predicha.

  • C, el conjunto de todas las clases.

  • pi(c), la probabilidad condicional de si la celda i contiene un objeto de la clase c∈C.

  • p^i(c), la probabilidad de clase condicional predicha.

En una ubicación, en la celda i, el modelo propone B candidatos de cuadros delimitadores y el que tiene más superposición con respecto a la gran verdad es el predictor "responsable". Fig. tomada de \cite{weng2017detection1}.

La función de pérdida solo penaliza el error de clasificación si un objeto está presente en la celda de la grilla, 1iobj=1. También solo penaliza el error de coordenadas del cuadro delimitador si ese predictor es "responsable" del cuadro de la gran verdad, 1ijobj=1.

Como detector de objetos de una etapa, YOLO es considerablemente rápido, pero no es bueno para reconocer objetos de forma irregular o un grupo de objetos pequeños debido a un número limitado de candidatos de cuadros delimitadores.

1.10 YOLOv2 / YOLO9000

El modelo YOLOv2 \cite{redmon2017yolo9000} es una versión mejorada de YOLO. YOLO9000 está construido sobre YOLOv2 pero entrenado con un dataset que combina el dataset de detección COCO y las 9000 clases principales de ImageNet.

1.10.1 YOLOv2

Se aplican una variedad de modificaciones para hacer que la predicción de YOLO sea más precisa y rápida, que incluyen:

  1. BatchNorm: añade una normalización por lotes (cada vez que nos referimos a batch normalization lo haremos usando la siguiente abreviatura: bn) en todas las capas convolucionales, lo que lleva a una mejora significativa sobre la convergencia.

  2. Importancia de la resolución de la imagen: aplicando la técnica de transferencia de conocimiento (fine-tuning) del modelo base con imágenes de alta resolución mejora el rendimiento en la detección.

  3. Detección de cajas ancla convolucional: en lugar de predecir las posiciones de los cuadros delimitadores usando capas completamente conectada sobre todo el mapa de características, YOLOv2 utiliza capas convolucionales para predecir las ubicaciones de las cajas ancla, como en una R-CNN más rápida. La predicción de las ubicaciones espaciales y las probabilidades de clase están desacopladas. En general, el cambio conduce a una ligera disminución en mAP, pero a un aumento en la memoria.

  4. Agrupamiento de K−mean de las diferentes dimensiones de caja; a diferencia de la R-CNN más rápida que utiliza tamaños de cajas ancla seleccionados a mano, YOLOv2 ejecuta el agrupamiento de k−mean en los datos de entrenamiento para encontrar las dimensiones de la caja ancla. La métrica de la distancia está diseñada para depender de las puntuaciones de IoU:

dist(x,ci)=1−IoU(x,ci),i=1,…,k

donde x es un candidato de cuadro de la gran verdad y ci es uno de los centroides. El número de centroides (cajas ancla) k se puede elegir mediante el método de elbow.

Las cajas anclas generadas por el agrupamiento proporcionan una mejor IoU promedio condicionada a un número fijo de cajas.

  1. Predicción de ubicación directa: YOLOv2 formula la predicción del cuadro delimitador de manera que no debería diverga demasido de la ubicación central. Si la predicción de la ubicación de la caja puede colocar la caja en cualquier parte de la imagen, como en la RPN, el entrenamiento del modelo podría volverse inestable.

Dada una caja ancla de tamaño (pw,ph) en la celda de la grilla con su esquina superior izquierda (cx,cy), el modelo predice el desplazamiento y la escala, (tx,ty,tw,yh) y el cuadro delimitador predicho correspontiene b tiene centro (bx,by) y tamaño (bw,bh). La puntuación de confianza es la función sigmoide (σ) de otra salida.

\begin{aligned} b_x &= \sigma(t_x) + c_x\ b_y &= \sigma(t_y) + c_y\ b_w &= p_w e^{t_w}\ b_h &= p_h e^{t_h}\ \text{Pr}(\text{object}) &\cdot \text{IoU}(b, \text{object}) = \sigma(t_o) \end{aligned}

Predicción de la ubicación del cuadro delimitador en el modelo YOLOv2. Fig. tomada de \cite{weng2017detection1}.

  1. Se agregan características detalladas (fine-grained features): YOLOv2 agrega una capa de paso para traer las características detalladas de una capa anterior a la última capa de salida. El mecanismo de esta capa de paso es similar a las asignaciones de identidad del modelo ResNet para extraer las características de mayor dimensión de las capas anteriores. Esto conduce a un aumento del rendimiento del 1%.

  2. Entrenamiento Multiescala: para entrenar el modelo para que sea más robusta a imágenes de entrada de tamaños diferentes, se muestrea aleatoriamente un nuevo tamaño de dimensión de entrada cada 10 lotes. Dado que las capas convolucionales de YOLOv2 reducen la muestra de la dimensión de entrada por un factor de 32, el nuevo tamaño muestreado es un múltiplo de 32.

  3. Modelo base ligero: para que las predicciones sean más rápidas, YOLOv2 adopa un modelo base ligero, DarkNet-19, que tiene 19 capas convolucionales y 5 capas de max-pooling. El punto clave es insertar una avg-pool y filtros convolucionales 1×1 entre capas convolucionales 3×3.

1.10.2 YOLO9000

Debido a que dibujar cuadros delimitadores en imágenes para la detección de objetos es computacionalmente costos que etiquetar las imágenes para su clasificación, el artículo origanl propone una forma para combinar un datset de detección de objetos pequeño con un dataset grande como ImageNet. El nombre YOLO9000 proviene de las 9000 clases principales de ImageNet. Durante el entrenamiento conjunto, si una imagen de entrada viene del dataset de clasificación, solo retropropaga la pérdida de la clasificación.

El dataset de detección tiene muchas menos etiquetas y más generales, ademas, las etiquetas cruzan múltiples datasets que a menudo no se exluyen mutuamente. Por ej., ImageNet tiene la etiqueta "Persian cat" mientras que en COCO la misma imagen puede ser etiquetada como "cat". Sin exclusividad mutua, no tiene sentido aplicar softmax sobre todas las clases.

Para realizar una mezcla eficiente de las etiquetas de ImageNet (1000 clases, de grano fino) con COCO/PASCAL (<100 clases, grano grueso), YOLO9000 construye una estructura de árbol jerárquico con referencia a WordNet para que las etiquetas generales estén más cerca de la raíz y las etiquetas de clase detalladas sean las hojas. De esta forma, "cat" es el nodo padre de "Persian cat".

La jerarquía de WordTree combina etiquetas de COCO e ImageNet. Los nodos azules son etiquetas de COCO y los nodos rojos son etiquetas de ImageNet. Fig. tomada de \cite{weng2017detection1}.

Para predecir la probabilidad de un nodo de clase, podemos seguir el camino desde el nodo hasta la ráiz:

Pr("persian cat" | contain a "physical object") = \
    Pr("persian cat" | "cat")
    Pr("cat" | "animal")
    Pr("animal" | "pysical object")
    Pr(contain a "physical object") # puntuación de confianza

Cabe destacar que Pr(contain a "physical object") es la puntuación de confianza predecida por separado en la estructura de pipeline de cuadro delimitador. El camino de la predicción de la probabilidad condicional puede detenerse en cualquier paso, según las etiquetas disponibles.

1.11 RetinaNet

El modelo de RetinaNet \cite{lin2017focal} es un detector de objetos denso de una etapa. Dos bloques de construcción cruciales son: (1) la pirámide de imágenes destacada y (2) el uso de la pérdida focal.

La pirámide de imágenes se basa en la idea intuitiva de que mapas de características de grano fino grandes en niveles anteriores son buenos para captura objetos pequeños y los mapas de características de grano grueso pequeños pueden detectar bien objetos grandes.

Ejemplo de pirámide de imágenes del modelo SSD (previamente descrito), donde la detección ocurre en cada capa piramidal, apuntando a objetos de varios tamaños. Fig. tomada de \cite{weng2017detection1}.

1.11.1 Pérdida Focal (Focal Loss)

Un problema para el entrenamiento de los modelos de detección de objetos es un desequilibrio extremo entre el fondo que no contiene ningún objeto y el primer plano que contiene objetos de interés. La pérdida focal está diseñada para asignar más peso a ejemplos "duros" que se clasifican erróneamente fácilmente (es decir, fondo con textura ruidosa u objetos parciales) y para reducir el peso de los ejemplos fáciles (es decir, fondo obviamente vacío).

Comenzando con una pérdida de entropía cruzada normal para una clasificación binaria,

CE(p,y)=−ylogp−(1−y)log(1−p),

donde y∈{0,1} es una etiqueta de la gran verdad, indicando si un cuadro delimitador contiene un objeto, y p∈[0,1] es la probabilidad predicha de objetividad (también conocida como puntuación de confianza).

Por conveniencia de notación, sea

pt={pif y=11−potherwise,CE(p,y)=CE(pt)=−logpt

Los ejemplos fácilmente clasificables con pt≫0.5 grande, es decir, cuando p está muy cerca de 0 (cuando y=0) o 1 (cuando y=1), pueden incurrir en una pérdida de magnitud no trivial. La pérdida focal agrega explícitamente un factor de ponderación (1−pt)γ,γ≥0 para cada término en la entropía cruzada, de modo que el peso es pequeño cuando pt es grande y, por lo tanto, los ejemplos fáciles se ponderan hacia abajo.

FL(pt)=−(1−pt)γlogpt

La pérdida focal se centra menos en ejemplos sencillos con un factor de (1−pt)γ. Fig. tomada de \cite{weng2017detection1}.

Para un mejor control en la dimensión de la función de ponderación, RetinaNet usa una variante de pérdida focal balanceada en α, donde α=0.25 y γ=2 funcionan mejor,

FL(pt)=−α(1−pt)γlogpt

Gráfico de los pesos de la pérdida focal α(1−pt)γ en función de pt dados diferentes valores de α y γ. Fig. tomada de \cite{weng2017detection1}

1.11.2 Pirámide de Imágenes Destacadas

La Pirámide de Imágenes Destacadas \cite{lin2017feature} es la red principa de RetinaNet. Siguiendo el mismo enfoque de la pirámide en SSD, las pirámides destacadas proporcionan un componente de visión básico para la detección de objetos a diferentes escalas.

La idea principal de la red de pirámide destacada se muestra en la fig. de abajo. La estructura base contiene una secuencia de niveles de pirámides, cada uno correspondiente a una etapa de la red. Una etapa contiene múltiple capas convolucionales del mismo tamaño y los tamaños de las estapas se reducen por un factor de 2. Sea la última capa de la i−ésima etapa como Ci.

Pirámide de Imágenes destacadas. Fig. tomada de \cite{weng2017detection1}.

Dos vías conectan las capas convolucionales:

  1. Camino de abajo hacia arriba: es el cálculo hacia adelante normal.

  2. Camino de arriba hacia abajo: va en dirección inversa, agregando mapas de características semánticamente más fuertes de regreso a los niveles de pirámide anteriores de un tamaño más grande a través de conexiones laterales.

*    **Primero, las características de un nivel más alto se muestrean espacialmente más gruesas para que sean 2 veces más grandes. Para la mejora de la imagen, el artículo original utiliza el muestreo superio del vecino más cercano. Si bien existen muchos algoritmos de escalado de imágenes, como el uso de la deconvolución, la adopción de otro método de escalado de imágenes podría o no mejorar el rendimiento de RetinaNet.**

*    **El mapa de características más grande se somete a una capa de convolución de 1×1 para reducir la dimensión del canal.**  

*    **Finalmente, estos dos mapas de características se mezclan mediante la agregación de elementos inteligente.**

*    **Las conexiones laterales solo ocurren en la última capa en las etapas, denotado como {Ci}, y el proceso continúa hasta que se genera el mapa de características combinado más fino (más grande). La predicción se realiza a partir de cada mapa mezclado después de una capa de convolución de 3×3, {Pi}.**

Según los estudios, el rango de importancia de los componentes del diseño piramidal de la imagen destacada es el siguiente: conexión lateral 1×1> detección de objetos en varias capas > enriquecimiento de arriba hacia abajo > representación piramidal (en comparación con comprobar solo la capa más fina).

1.11.3 Arquitectura del Modelo

La pirámide destacada se construye sobre la arquitectura ResNet. Donde ResNet tiene 5 bloques convolucionales (= etapas de red -niveles de pirámide-). La última capa del i−ésimo nivelo, Ci, tiene una resolución 2i más baja que la dimensión de entrada sin procesar.

RetinaNet utiliza los niveles de pirámide de características P3 a P7:

  • P3 a P5 son calculados de la etapa residual correspondiente a ResNet de C3 a C5. Estos están conectados por vías de arriba hacia abajo y de abajo hacia arriba.

  • P6 se obtiene a través de una convolución 3×3, stride 2 en la parte superior de C5.

  • P7 aplica la función de activación ReLU y una convolución de 3×3, stride 2 en la parte superior de P6.

Añadiendo niveles piramidales más altos sobre ResNet mejora el rendimiento para la detección de objetos más grandes.

Al igual que en la arquitectura SSD, la detección ocurre en todos los niveles de la pirámide al hacer una predicción a partir de cada mapa de características combinado. Debido a que las predicciones comparten el mismo clasificador y el regresor de caja, todas se forman para tener la misma dimensión de canal d=256.

Hay A=9 cajas ancla por nivel:

  • El tamaño base corresponde a las áreas de 322 a 5123 píxeles en P3 a P7 respectivamente. Hay tres relaciones de tamaños: {20,21/3,22/3}.

  • Para cada tamaño, hay tres relaciones de aspectos: {1/2,1,2}.

Como es habitual, para cada caja ancla, el modelo genera una probabilidad de clase para cada una de las K clases en la clasificación de la subred y retrocede el desplazamiento desde la caja ancla al objeto de la gran verdad más cercana en la subred de la caja de regresión. La subred de clasificación adopta la pérdida focal (vista anteriormente).

Arquitectura del modelo RetinaNet que utiliza una red troncal FPN sobre ResNet. Fig. tomada de \cite{weng2017detection1}.

1.12 YOLOv3

YOLOv3 se crea aplicando técnicas de diseño de YOLOv3. La lista de mejoras se muestra a continuación:

  1. Regresión logística para las puntuaciones de confianza: YOLOv3 predice una puntuación de confianza para cada cuadro delimitador usando la regresión logística, mientras que YOLO y YOLOv2 usan la suma de los errores al cuadrado para los términos de clasificación. La regresión lineal de la predicción del desplazamiento conduce a una disminución de mAP.

  2. No más softmax para la predicción de clase: cuando se predice la confianza de clase, YOLOv3 utiliza clasificadores logísticos independientes múltiples para cada clase en lugar de una sola capa softmax. Esto es una mejora notable, especialmente cuando se considera que una imagen puede tener múltiples etiquetas y no se garantiza que todas las etiquetas sean mutuamente excluyentes.

  3. DarkNet + ResNet como el modelo base: el nuevo modelo de DarkNet-53 todavía se base en capas convolucionales sucesivas de 3×3 y 1×1, al igual que la arquitectura DarkNet original, pero se le agregan bloques residuales.

  4. Predicción multiescala: inspirado en la pirámide de imágenes, YOLOv3 agrega varias capas convolucionales después del modelo extractor y hace predicciones en tres escalas diferentes entre estas capas de convolución. De esta manera, tiene que lidear con muchos más candidatos a cuadros delimitadores de varios tamaños en general.

  5. Concatenación de salto de capa: YOLOv3 también agrega conexiones cruzadas entre dos capas de predicción (excepto para la capa de salida) y los mapas de características anteriores. El modelo primero toma muestras de los mapas de características gruesas y luego las mezcla con las características anteriores mediante la concatenación. La combinación con información de grano más fino lo hace mejor para detectar objetos pequeños.

Como dato curioso, la pérdida focal no ayuda a YOLOv3, posiblemente se deba al uso de λnoobj y λcoord, ya que estos incrementan la pérdida de las predicciones de las ubicaciones de los cuadros delimitadores y disminuyen la pérdida de las predicciones de confianza para las cajas de fondo.

Comparación de varios modelos rápidos de detección de objetos y el rendimiento mAP. Fig. tomada de \cite{weng2017detection4}.
