def plot(X, Y=None, xlabel=None, ylabel=None, legend=None, xlim=None,
         ylim=None, xscale='linear', yscale='linear',
         fmts=('-', 'm--', 'g-.', 'r:'), figsize=(3.5, 2.5), axes=None):
    """Plot data points.

    Defined in :numref:`sec_calculus`"""
    if legend is None:
        legend = []

    set_figsize(figsize)
    axes = axes if axes else plt.gca()

    # Return True if `X` (tensor or list) has 1 axis
    def has_one_axis(X):
        return (hasattr(X, "ndim") and X.ndim == 1 or isinstance(X, list)
                and not hasattr(X[0], "__len__"))

    if has_one_axis(X):
        X = [X]
    if Y is None:
        X, Y = [[]] * len(X), X
    elif has_one_axis(Y):
        Y = [Y]
    if len(X) != len(Y):
        X = X * len(Y)
    axes.cla()
    for x, y, fmt in zip(X, Y, fmts):
        if len(x):
            axes.plot(x, y, fmt)
        else:
            axes.plot(y, fmt)
    set_axes(axes, xlabel, ylabel, xlim, ylim, xscale, yscale, legend)

def show_images(imgs, num_rows, num_cols, titles=None, scale=1.5):
    """Plot a list of images.

    Defined in :numref:`sec_fashion_mnist`"""
    figsize = (num_cols * scale, num_rows * scale)
    _, axes = plt.subplots(num_rows, num_cols, figsize=figsize)
    axes = axes.flatten()
    for i, (ax, img) in enumerate(zip(axes, imgs)):
        if torch.is_tensor(img):
            # Tensor Image
            ax.imshow(img.numpy())
        else:
            # PIL Image
            ax.imshow(img)
        ax.axes.get_xaxis().set_visible(False)
        ax.axes.get_yaxis().set_visible(False)
        if titles:
            ax.set_title(titles[i])
    return axes

class Accumulator:
    """For accumulating sums over `n` variables."""
    def __init__(self, n):
        """Defined in :numref:`sec_softmax_scratch`"""
        self.data = [0.0] * n

    def add(self, *args):
        self.data = [a + float(b) for a, b in zip(self.data, args)]

    def reset(self):
        self.data = [0.0] * len(self.data)

    def __getitem__(self, idx):
        return self.data[idx]

def accuracy(y_hat, y):
    """Compute the number of correct predictions.

    Defined in :numref:`sec_softmax_scratch`"""
    if len(y_hat.shape) > 1 and y_hat.shape[1] > 1:
        y_hat = argmax(y_hat, axis=1)
    cmp = astype(y_hat, y.dtype) == y
    return float(reduce_sum(astype(cmp, y.dtype)))


def evaluate_accuracy(net, data_iter):
    """Compute the accuracy for a model on a dataset.

    Defined in :numref:`sec_softmax_scratch`"""
    if isinstance(net, torch.nn.Module):
        net.eval()  # Set the model to evaluation mode
    metric = Accumulator(2)  # No. of correct predictions, no. of predictions

    with torch.no_grad():
        for X, y in data_iter:
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def evaluate_loss(net, data_iter, loss):
    """Evaluate the loss of a model on the given dataset.

    Defined in :numref:`sec_model_selection`"""
    metric = Accumulator(2)  # Sum of losses, no. of examples
    for X, y in data_iter:
        out = net(X)
        y = reshape(y, out.shape)
        l = loss(out, y)
        metric.add(reduce_sum(l), size(l))
    return metric[0] / metric[1]

def try_gpu(i=0):
    """Return gpu(i) if exists, otherwise return cpu().

    Defined in :numref:`sec_use_gpu`"""
    if torch.cuda.device_count() >= i + 1:
        return torch.device(f'cuda:{i}')
    return torch.device('cpu')

def try_all_gpus():
    """Return all available GPUs, or [cpu(),] if no GPU exists.

    Defined in :numref:`sec_use_gpu`"""
    devices = [torch.device(f'cuda:{i}')
             for i in range(torch.cuda.device_count())]
    return devices if devices else [torch.device('cpu')]

def evaluate_accuracy_gpu(net, data_iter, device=None):
    """Compute the accuracy for a model on a dataset using a GPU.

    Defined in :numref:`sec_lenet`"""
    if isinstance(net, nn.Module):
        net.eval()  # Set the model to evaluation mode
        if not device:
            device = next(iter(net.parameters())).device
    # No. of correct predictions, no. of predictions
    metric = Accumulator(2)

    with torch.no_grad():
        for X, y in data_iter:
            if isinstance(X, list):
                # Required for BERT Fine-tuning (to be covered later)
                X = [x.to(device) for x in X]
            else:
                X = X.to(device)
            y = y.to(device)
            metric.add(accuracy(net(X), y), size(y))
    return metric[0] / metric[1]

def train_ch6(net, train_iter, test_iter, num_epochs, lr, device):
    """Train a model with a GPU (defined in Chapter 6).

    Defined in :numref:`sec_lenet`"""
    def init_weights(m):
        if type(m) == nn.Linear or type(m) == nn.Conv2d:
            nn.init.xavier_uniform_(m.weight)
    net.apply(init_weights)
    print('training on', device)
    net.to(device)
    optimizer = torch.optim.SGD(net.parameters(), lr=lr)
    loss = nn.CrossEntropyLoss()
    animator = Animator(xlabel='epoch', xlim=[1, num_epochs],
                            legend=['train loss', 'train acc', 'test acc'])
    timer, num_batches = Timer(), len(train_iter)
    for epoch in range(num_epochs):
        # Sum of training loss, sum of training accuracy, no. of examples
        metric = Accumulator(3)
        net.train()
        for i, (X, y) in enumerate(train_iter):
            timer.start()
            optimizer.zero_grad()
            X, y = X.to(device), y.to(device)
            y_hat = net(X)
            l = loss(y_hat, y)
            l.backward()
            optimizer.step()
            with torch.no_grad():
                metric.add(l * X.shape[0], accuracy(y_hat, y), X.shape[0])
            timer.stop()
            train_l = metric[0] / metric[2]
            train_acc = metric[1] / metric[2]
            if (i + 1) % (num_batches // 5) == 0 or i == num_batches - 1:
                animator.add(epoch + (i + 1) / num_batches,
                             (train_l, train_acc, None))
        test_acc = evaluate_accuracy_gpu(net, test_iter)
        animator.add(epoch + 1, (None, None, test_acc))
    print(f'loss {train_l:.3f}, train acc {train_acc:.3f}, '
          f'test acc {test_acc:.3f}')
    print(f'{metric[2] * num_epochs / timer.sum():.1f} examples/sec '
          f'on {str(device)}')

class Animator:
    """For plotting data in animation."""
    def __init__(self, xlabel=None, ylabel=None, legend=None, xlim=None,
                 ylim=None, xscale='linear', yscale='linear',
                 fmts=('-', 'm--', 'g-.', 'r:'), nrows=1, ncols=1,
                 figsize=(3.5, 2.5)):
        """Defined in :numref:`sec_softmax_scratch`"""
        # Incrementally plot multiple lines
        if legend is None:
            legend = []
        use_svg_display()
        self.fig, self.axes = plt.subplots(nrows, ncols, figsize=figsize)
        if nrows * ncols == 1:
            self.axes = [self.axes, ]
        # Use a lambda function to capture arguments
        self.config_axes = lambda: set_axes(
            self.axes[0], xlabel, ylabel, xlim, ylim, xscale, yscale, legend)
        self.X, self.Y, self.fmts = None, None, fmts

    def add(self, x, y):
        # Add multiple data points into the figure
        if not hasattr(y, "__len__"):
            y = [y]
        n = len(y)
        if not hasattr(x, "__len__"):
            x = [x] * n
        if not self.X:
            self.X = [[] for _ in range(n)]
        if not self.Y:
            self.Y = [[] for _ in range(n)]
        for i, (a, b) in enumerate(zip(x, y)):
            if a is not None and b is not None:
                self.X[i].append(a)
                self.Y[i].append(b)
        self.axes[0].cla()
        for x, y, fmt in zip(self.X, self.Y, self.fmts):
            self.axes[0].plot(x, y, fmt)
        self.config_axes()
        display.display(self.fig)
        display.clear_output(wait=True)

class Timer:
    """Record multiple running times."""
    def __init__(self):
        """Defined in :numref:`subsec_linear_model`"""
        self.times = []
        self.start()

    def start(self):
        """Start the timer."""
        self.tik = time.time()


    def stop(self):
        """Stop the timer and record the time in a list."""
        self.times.append(time.time() - self.tik)
        return self.times[-1]


    def avg(self):
        """Return the average time."""
        return sum(self.times) / len(self.times)


    def sum(self):
        """Return the sum of time."""
        return sum(self.times)


    def cumsum(self):
        """Return the accumulated time."""
        return np.array(self.times).cumsum().tolist()

1 Visión por Computadora

En los último años, el deep learning ha sido el poder transformador para mejorar el rendimiento de los sistemas de visión por computadora. Se puede decir que las aplicación de visión artificial más avanzandas son casi inseparables del deep learning. Por ende, en las siguientes secciones veremos métodos y aplicaciones que han sido influyentes recientemente en la academia y en la industria.

1.1 Aumento de Imagen (Image Augmentation)

La Aumento de imagen (image augmentation) genera ejemplos de entrenamiento similares pero distintos después de una serie de cambios aleatorios en las imágenes de entrenamiento, expandiendo así el tamaño del conjunto de entrenamiento. Alternativamente, el aumento de imágenes puede estar motivado por el hecho de que los ajustes aleatorios de los ejemplos de etrenamiento permiten que los modelos dependan menos de ciertos atributos, mejorando así su capacidad de generalización. Por ej., podemos recortar una imagen de diferentes formas para que el objeto de interés aparezca en diferentes posiciones, reduciendo la dependencia de un model sobre la posición del objeto. Podemos también ajustar factores como el brillo y el color para reducir la sensibilidad del modelo hacia los colores.*

1.1.1.1 Volteado y Recortado (Flipping and Cropping)

Voltear la imagen a la derecha o izquierda por lo general no cambia la categoría del objeto. Este es uno de los primeros y más utilizados métodos de aumento de imágenes. A continuación, usamos el módulo de transformaciones de pytorch para crear una instancia de RandomFlipLeftRight, que voltea una imagen de izquierda a derecha con un 50% de probabilidad.

Código Python · celda 18
apply(
    img, 
    torchvision.transforms.RandomHorizontalFlip() # aug
)

El volteado hacia arriba o hacia abajo no es tan común como el volteado hacia la izquierda y derecha.

En el ejemplo, la imagen del gato está en el medio de la imagen, pero este puede no ser el caso general. Anteriormente, explicamos que la capa de max-pool puede reducir la sensibilidad de una capa de convolución sobre una posición objetivo. Además, podemos recortar aleatoriamente la imagen para hacer que los objetos aparezcan en diferentes posiciones en la imagen a diferentes escalas, lo que también puede reducir la sensibilidad de un modelo a la posición de destino.

En el siguiente código, recortamos aleatoriamente un área de 10%\ 100% del área original cada vez, y la relación entre el ancho y el alto de esta area es seleccionada aleatoriamente de 0.5\ 2. Luego, el ancho y el alto de la región se escalan a 200 píxeles. A menos que se especifique lo contrario, el número aleatorio entre a y b se refiere al valor continuo obtenido por una distribución aleatorio uniforme del intervalo [a,b].

    size    = (200, 200),

    ratio   = (0.5, 2)

class Residual(nn.Module):  
    """The Residual block of ResNet."""

    def __init__(
        self, 
        input_channels, 
        num_channels,
        use_1x1conv         = False, 
        strides             = 1
    ):
        super().__init__()

        # Capa convolucional 1 de 3 x 3, padding 1, stride = strides = 1 (por defecto)
        self.conv1 = nn.Conv2d(
            in_channels     = input_channels, 
            out_channels    = num_channels,
            kernel_size     = 3, 
            padding         = 1, 
            stride          = strides
        )

        # Capa convolucional 2 de 3 x 3, padding 1, stride = strides = 1 (por defecto)
        # input_channels = output_channels = num_channels
        self.conv2 = nn.Conv2d(
            in_channels     = num_channels, 
            out_channels    = num_channels, 
            kernel_size     = 3, 
            padding         = 1
        )   

        # Se utiliza en caso de que el número de canales sea diferente, por eso se añade una capa convolucional
        # extra de 1 x 1
        if use_1x1conv:
            self.conv3 = nn.Conv2d(
                in_channels = input_channels, 
                out_channels= num_channels,
                kernel_size =1, 
                stride      = strides
            )
        else:
            self.conv3 = None

        # Capas de batch normalization
        self.bn1            = nn.BatchNorm2d(num_channels)
        self.bn2            = nn.BatchNorm2d(num_channels)

    def forward(self, X):
        # Después de aplicar BN a la capa convolucional1, se aplica la función de activación ReLu
        Y                   = F.relu(self.bn1(self.conv1(X)))
        # Luego, se aplica BN a la capa convolucional 2 (que recibe como entrada la salida de la capa convolucional 1
        # luego de aplicar bn1 y Relu)
        Y                   = self.bn2(self.conv2(Y))

        # se realiza la convolución 3 si es que se definió.
        if self.conv3:
            X               = self.conv3(X)


        Y                  += X
        return F.relu(Y)

def resnet18(num_classes):
    """A slightly modified ResNet-18 model.

    Defined in :numref:`sec_multi_gpu_concise`"""
    def resnet_block(num_channels, num_residuals, first_block=False):
        blk = nn.Sequential()
        for i in range(num_residuals):
            if i == 0 and not first_block:
                blk.add(Residual(
                    num_channels, use_1x1conv=True, strides=2))
            else:
                blk.add(Residual(num_channels))
        return blk

    net = nn.Sequential()
    # This model uses a smaller convolution kernel, stride, and padding and
    # removes the maximum pooling layer
    net.add(nn.Conv2D(64, kernel_size=3, strides=1, padding=1),
            nn.BatchNorm(), nn.Activation('relu'))
    net.add(resnet_block(64, 2, first_block=True),
            resnet_block(128, 2),
            resnet_block(256, 2),
            resnet_block(512, 2))
    net.add(nn.GlobalAvgPool2D(), nn.Dense(num_classes))
    return net

1.2 Fine-Tuning

Supongamos que queremos reconocer diferentes tipos de sillas a partir de imágenes y luego hacer una recomendación de enlaces de compra a los usuarios. Un método posible es identificar primero 100 sillas comunes, tomar 1000 imágenes de diferentes ángulas para cada silla y luego entrenar un modelo de clasificación en el dataset de imágenes recopilado.

Sin embargo, en comparación con el número de imágenes del dataset ImageNet, sigue siendo muy pequeño. Esto puede llevar a un sobreajust de modelos complicados que son adecuados para ImageNet en este datset de sillas. Además, debido a la cantidad limitad de ejemplos de entrenamiento, es posible que la presición del model no cumpla con los requisitos prácticos.

Para resolver los problemas mencionados, una solución obvia es recolectar más datos. Sin embargo, conseguir y etiquetar los datos puede tomar mucho tiempo y dinero. Por ej., para ImageNet, los investigadores gastaron millones de dólares en fondos de investigación. Aunque el costo actual de recopilación de datos se ha reducido significativamente, este costo aún no se puede ignorar.

Otra solución es aplicar transferencia de conocimiento (trasnfer learning) para transferir el conocimiento aprendido desde el dataset fuente hacia el dataset objetivo. Por ej., aunque la mayoría de las imágenes en el dataset ImageNet no tienen nada que ver con las sillas, el modelo entrenado en este dataset puede extraer características de imágenes más generales, que pueden ayudar a identificar bordes, texturas, formas y composición de objetos. Estas características similares también pueden ser efectivas para reconocer sillas.

Dentro del transfer learning, la técnica de fine-tunning consiste en realizar pequeños ajustes para obtener un comportamiento notablemente mejor.

1.2.1 Pasos

En la siguiente sección, introducimos una de las técnicas más comunes en la transferencia de conocimiento (transfer learning): fine-tunning, que consiste en 4 pasos:

  1. Pre-entrenar un modelo de red neuronal, es decir, entrenar el modelo origen, en un dataset de origen (por ej., el dataset ImageNet).
  2. Crear un nuevo modelo de red neuronal, es decir, el modelo objetivo. Esto copia todo los diseños del modelo y sus parámetros en el modelo origen, excepto la capa de salida. Suponemos que estos parámetros contienen el conocimiento aprendido del dataset de origen y este conocimiento también será aplicado al dataset objetivo. También, se asume que la capa de salida del modelo de origen está estrechamente relacionada con las etiquetas del dataset de origen; por tanto, no se utiliza en el modelo de destino.
  3. Añadir una capa de salida al modelo objetivo, cuyo número de salidas es el número de las categorías del dataset objetivo. Luego, se inicializa aleatoriamente los parámetros del modelo para esta capa.
  4. Entrenar el modelo objetivo sobre el dataset objetivo, es decir, sobre el dataset de sillas. La capa de salida será entrenada desde cero, mientras que los parámetros de todas las otras capas se ajustarán (fine-tuned) en función de los parámetros del modelo de origen.

Cuando los datasets destino son mucho más pequeños que los datasets de origen, fine-tunning ayuda a mejorar la capacidad de generalización de los modelos.

1.3.1 Cuadros delimitadores (Bounding Boxes)

En la detección de objetos, por lo general se utilizan cuadtros delimitadores (bounding boxes) para describir la ubicación espacial de un objeto. El cuadro delimitador es rectangular, que está determinado por las coordenadas x e y de la esquina superior izquierda del rectángulo y las coordenadas de la esquina inferior derecha. Otra representación del cuadro delimitador de uso común son las coordenadas del eje (x,y) del centro del cuadro delimitador y el ancho y alto del cuadro.

Aquí, definimos funciones para convertir estas dos representaciones: box_corner_to_center convierte de la representación de dos esquinas a la representación de centro-ancho-alto, y box_center_to_corner viceversa. Los cuadros de arumentos de entrada deben ser un tensor de forma bidimensional (n,4), donde n es el número de cuadros delimitadores.

Código Python · celda 77
def box_corner_to_center(boxes):
    """Convert from (upper-left, lower-right) to (center, width, height)."""

    x1, y1, x2, y2  = boxes[:, 0], boxes[:, 1], boxes[:, 2], boxes[:, 3]
    cx              = (x1 + x2) / 2
    cy              = (y1 + y2) / 2
    w               = x2 - x1
    h               = y2 - y1
    boxes           = torch.stack((cx, cy, w, h), axis=-1)

    return boxes


def box_center_to_corner(boxes):
    """Convert from (center, width, height) to (upper-left, lower-right)."""
    
    cx, cy, w, h    = boxes[:, 0], boxes[:, 1], boxes[:, 2], boxes[:, 3]
    x1              = cx - 0.5 * w
    y1              = cy - 0.5 * h
    x2              = cx + 0.5 * w
    y2              = cy + 0.5 * h
    boxes           = torch.stack((x1, y1, x2, y2), axis=-1)

    return boxes

1.4.1 Generando múltiples cajas de anclaje

Supongamos que la imagen de entrada tiene una altura h y un ancho w. Luego, generamos cajas de anclaje con diferentes formas centradas sobre cada píxel de la imagen. Sea la escala s∈(0,1] y sea la relación de aspecto aspect ratio (relación de ancho y alto) r>0. Entonces el ancho y el alto de la caja de anclaje son wsr y hsr respectivamente. Cuando se da la posición central, se determina una caja de anclaje con ancho y alto conocidos.

Para generar múltiples cajas de anclajes con diferentes formas, establecemos una serie de escalas s1,…,sn y una serie de relaciones de aspectos r1,…,rm. Cuando usamos todas las combinaciones de estas escalas y las relaciones de aspectos con cada píxel como centro, la imagen de entrada tendrá un total de whnm cajas de anclaje. Aunque estas cajas de anclaje puede cubrir todos los cuadros delimitadores de la "gran verdad (ground-truth)", la complejidad computacional es muy alta. En la práctica, sólo podemos considerar aquellas combinaciones que contienen s1 o r1:

(s1,r1),(s1,r2),…,(s1,rm),(s2,r1),(s3,r1),…,(sn,r1).

Es decir, el número de cajas de anclaje centradas en el mismo píxel es n+m−1. Para la imagen de entrada completa, generamos un total de wh(n+m−1) cajas de anclaje.

El método anterior de generar cajas de anclaje se implementa a continuación mediante la función multibox_prior. Específicamo la imagen de entrada, una lista de escalas, y una lista de relaciones de aspecto, luego esta función devolverá todas las cajas de anclaje.

def multibox_prior(data, sizes, ratios):
    """Generate anchor boxes with different shapes centered on each pixel."""
    
    print(f"data.shape: {data.shape}")

    in_height, in_width             = data.shape[-2:]
    device, num_sizes, num_ratios   = data.device, len(sizes), len(ratios)
    boxes_per_pixel                 = (num_sizes + num_ratios - 1)
    size_tensor                     = torch.tensor(sizes, device=device)
    ratio_tensor                    = torch.tensor(ratios, device=device)

    # Offsets are required to move the anchor to the center of a pixel. Since
    # a pixel has height=1 and width=1, we choose to offset our centers by 0.5
    offset_h, offset_w              = 0.5, 0.5
    steps_h                         = 1.0 / in_height  # Scaled steps in y axis
    steps_w                         = 1.0 / in_width  # Scaled steps in x axis

    # Generate all center points for the anchor boxes
    center_h                        = (torch.arange(in_height, device=device) + offset_h) * steps_h
    center_w                        = (torch.arange(in_width, device=device) + offset_w) * steps_w
    shift_y, shift_x                = torch.meshgrid(center_h, center_w)
    shift_y, shift_x                = shift_y.reshape(-1), shift_x.reshape(-1) # size = in_height * in_width
    
    print(f"center_h: {center_h.shape} \n center_w: {center_w.shape} \n")
    print(f"shift_y: {shift_y.shape} \n shift_x: {shift_x.shape} \n")

    # Generate `boxes_per_pixel` number of heights and widths that are later
    # used to create anchor box corner coordinates (xmin, xmax, ymin, ymax)
    w = torch.cat((size_tensor * torch.sqrt(ratio_tensor[0]),
                   sizes[0] * torch.sqrt(ratio_tensor[1:])))\
                   * in_height / in_width  # Handle rectangular inputs

    h = torch.cat(
        (
            size_tensor / torch.sqrt(ratio_tensor[0]),
            sizes[0] / torch.sqrt(ratio_tensor[1:])
        )
    )

    print(f"w: {w.shape} \n h: {h.shape} \n")


    # Divide by 2 to get half height and half width
    anchor_manipulations = torch.stack(
        (-w, -h, w, h)
    ).T.repeat(in_height * in_width, 1) / 2

    print(f"anchor_manipulations: {anchor_manipulations.shape} \n")

    # Each center point will have `boxes_per_pixel` number of anchor boxes, so
    # generate a grid of all anchor box centers with `boxes_per_pixel` repeats
    out_grid = torch.stack(
        [
            shift_x, 
            shift_y, 
            shift_x, 
            shift_y
        ],
        dim=1).repeat_interleave(boxes_per_pixel, dim=0)

    output = out_grid + anchor_manipulations

    return output.unsqueeze(0)

1.4.2 Intersección sobre Unión

Si se conoce el cuadro delimitador de la "gran verdad" del objeto, ¿cómo se podría cuantificar el "bien" aquí? Intuitivamente, podemos mediar la similitud entre la caja de anclaje y el cuadro delimitador de la gran verdad. Sabemos que el índice de Jaccard puede medir la similitud entre dos conjuntos. Dados los conjuntos A y B, su índice de Jaccard es el tamaño de su intersección dividido por el tamaño de su unión:

J(𝒜,ℬ)=|𝒜∩ℬ||𝒜∪ℬ|.

De hecho, podemos considerar el área de píxeles de cualquier cuadro delimitador como un conjunto de píxeles. De esta manera, podemos medir la similitud de dos cuadros delimitadores por el índice de Jaccard de sus conjuntos de píxeles. Para dos cuadros delimitadores, usualmente nos referimos a su índice de Jaccard como la intersección sobre la unión (IoU), que es la relación del area de su intersección y el área de su unión, como se muestra en la figura de abajo. El rango de una IoU está entre 0 y 1: 0 significa que dos cuadros delimitadores no se sobreponen en absoluto, mientras que 1 indica que los cuadros delimitadores son iguales.

IoU es la relación del área de intersección con el área de unión de dos cuadros delimitadores.

A lo largo de las siguientes secciones, utilizaremos IoU para medir la similitud entre las cajas de anclaje y los cuadros delimitadores de la gran verdad, y entre diferentes cajas de anclajes. Dadas dos lista de cajas de anclaje o delimitadores, la siguiente función box_iou calcula su IoU por pares en estas dos listas:

Código Python · celda 102
def box_iou(boxes1, boxes2):
    """Compute pairwise IoU across two lists of anchor or bounding boxes."""
    box_area            = lambda boxes: ((boxes[:, 2] - boxes[:, 0]) *
                              (boxes[:, 3] - boxes[:, 1]))
    
    # Shape of `boxes1`, `boxes2`, `areas1`, `areas2`: (no. of boxes1, 4),
    # (no. of boxes2, 4), (no. of boxes1,), (no. of boxes2,)
    areas1              = box_area(boxes1)
    areas2              = box_area(boxes2)

    # Shape of `inter_upperlefts`, `inter_lowerrights`, `inters`: (no. of
    # boxes1, no. of boxes2, 2)
    inter_upperlefts    = torch.max(boxes1[:, None, :2], boxes2[:, :2])
    inter_lowerrights   = torch.min(boxes1[:, None, 2:], boxes2[:, 2:])
    inters              = (inter_lowerrights - inter_upperlefts).clamp(min=0)

    # Shape of `inter_areas` and `union_areas`: (no. of boxes1, no. of boxes2)
    inter_areas         = inters[:, :, 0] * inters[:, :, 1]
    union_areas         = areas1[:, None] + areas2 - inter_areas
    
    return inter_areas / union_areas

1.4.3 Etiquetado de Cajas de Anclaje en Datos de Entrenamiento

En el dataset de entrenamiento, consideramos cada caja de anclaje como una muestra de entrenamiento. Para entrenar un modelo de detección de objetos, necesitamos clases y etiquetas de desplazamiento para cada caja de anclaje, donde el primero es la clase del objeto relevante para la caja de anclaje y el segundo es el desplazamiento del cuadro delimitador de la gran verdad con respecto a la caja de anclaje. Durante la predicción, para cada imagen generamos múltiples cajas de anclaje, predecimos clases y desplazamientos para todas las cajas de anclaje, ajustamos sus posiciones de acuerdo con los desplazamientos previstos para obtener los cuadros delimitadores predichos y, finalmente, solo generamos aquellos cuadros delimitadores que sastifcen ciertos criterios.

Como sabemos, el conjunto de entrenamiento de detección de objetos incluye etiquetas para las ubicaciones de los cuadros delimitadores de la gran verdad y clases de sus objetos delimitados. Para etiquetar cualquier caja de anclaje, nos referimos a la ubicación etiquetada y la clase de su cuadro delimitador de la gran verdad que está más cerca de la caja ancla.

A continuación, describimos un algoritmo para asignar cuadros delimitadores de la gran verdad más cernaos a cuadros de anclaje.

1.4.3.1 Asignar Cuadros Delimitadores de la Gran Verdad (Ground-Truth) a las Cajas de Anclaje

Dada una imagen, supongamos que las cajas de anclaje son A1,A2,…,Ana y los cuadros delimitadores de la gran verdad son B1,B2,…,Bnb, donde na≥nb. Sea una matriz 𝐗∈ℝna×nb, cuyos elementos xij en la i−ésima fila y en la j−ésima columna son la IoU de la caja de anclaje Ai y el cuadro delimitador de la gran verdad Bj. El algoritmo consiste en los siguientes pasos:

  1. Encontrar el elemento más grande en la matriz 𝐗 y denotando sus índices de de fila y columna como i1 y j1, respectivamente. Entonces, el cuadro delimitador de la gran verdad Bj1 es asignado a la caja de anclaje Ai1. Esto es bastante intuitivo porque Ai1 y Bj1 son los más cercanos entre todos los pares de cajas de anclaje y cuadros delimitadores de la gran verdad. Después de la primera asignación, se descartan todos los elementos de la fila i1 y la columna j1 de la matriz 𝐗.

  2. Encontrar los elementos restantes más grandes en la matriz 𝐗 y denotando sus índices de fila y columna como i2 y j2, respectivamente. Asignamos el cuadro delimitador de la gran verdad Bj2 a la caja de anclaje Ai2 y se descartan todos los elementos de la fila i2 y la columna j2 de la matriz 𝐗.

  3. En este punto, los elementos de dos filas y columnas de la matriz 𝐗 han sido descartados. Continuamos hasta que se descarten todos los lementos en nb columnas de la matriz 𝐗. Luego, hasta este momentto, se asignó un cuadro delimitador de la gran verdad a cada uno de las nb cajas de anclaje.

  4. Solo se atraviesa las cajas de anclaje na−nb restantes. Por ej., dada cualquier caja de anclaje Ai, se encuentra el cuadro delimitador de la gran verdad Bj con el IoU más grande con Ai a lo largo de la i−ésima fila de la matriz 𝐗, y se asignar Bj a Ai solo si este IoU es mayor que un umbral predefinido.

def assign_anchor_to_bbox(ground_truth, anchors, device, iou_threshold=0.5):
    """Assign closest ground-truth bounding boxes to anchor boxes."""

    num_anchors, num_gt_boxes       = anchors.shape[0], ground_truth.shape[0]

    # Element x_ij in the i-th row and j-th column is the IoU of the anchor
    # box i and the ground-truth bounding box j
    jaccard                         = box_iou(anchors, ground_truth)

    # Initialize the tensor to hold the assigned ground-truth bounding box for
    # each anchor
    anchors_bbox_map = torch.full(
                        (num_anchors, ), 
                        -1, 
                        dtype       = torch.long,
                        device      = device
                    )
    
    # Assign ground-truth bounding boxes according to the threshold
    max_ious, indices               = torch.max(jaccard, dim=1)
    anc_i                           = torch.nonzero(max_ious >= 0.5).reshape(-1)
    box_j                           = indices[max_ious >= 0.5]
    anchors_bbox_map[anc_i]         = box_j

    col_discard                     = torch.full((num_anchors,), -1)
    row_discard                     = torch.full((num_gt_boxes,), -1)

    for _ in range(num_gt_boxes):
        max_idx                     = torch.argmax(jaccard)  # Find the largest IoU
        box_idx                     = (max_idx % num_gt_boxes).long()
        anc_idx                     = (max_idx / num_gt_boxes).long()

        anchors_bbox_map[anc_idx]   = box_idx
        jaccard[:, box_idx]         = col_discard
        jaccard[anc_idx, :]         = row_discard
    
    return anchors_bbox_map

1.4.3.3 Etiquetado de Clases y Desplazamientos

Ahora podemos etiqueta las clases y desplazamientos para cada caja de anclaje. Supongamo que la caja de anclaje A es asignada al cuadro delimitador B de la gran verdad. Por una parte, la clase de la caja de anclaje A será etiquetada como la de B. Por otra parte, el desplazamiento de la caja de anclaje A será etiquetada de acuerdo a la posición relativa entre las coordenadas centras de B y A juntas con el tamaño relativo entre estas dos cajas. Dadas las diferentes posiciones y tamaños de diferentes cajas en el dataset, podemos aplicar transformaciones a esas posiciones relativas y tamaños que pueden conducir a compensaciones distriubidas de manera más uniforme que son más fáciles de ajustar.

Aquí, se describe una transformación común. Dadas las coordenadas centrales de A y de B como (xa,ya) y (xb,yb) y sus anchos y altos como wa y wb, ha y hb, respectivamente, podemos etiquetar el desplazamiento de A como

(xb−xawa−μxσx,yb−yaha−μyσy,logwbwa−μwσw,loghbha−μhσh),

donde los valores por defecto de las constantes son μx=μy=μw=μh=0,σx=σy=0.1 y σw=σh=0.2.

Código Python · celda 109
def offset_boxes(anchors, assigned_bb, eps=1e-6):
    """Transform for anchor box offsets."""

    c_anc           = box_corner_to_center(anchors)
    c_assigned_bb   = box_corner_to_center(assigned_bb)
    offset_xy       = 10 * (c_assigned_bb[:, :2] - c_anc[:, :2]) / c_anc[:, 2:]
    offset_wh       = 5 * torch.log(eps + c_assigned_bb[:, 2:] / c_anc[:, 2:])
    offset          = torch.cat([offset_xy, offset_wh], axis=1)
    
    return offset

def multibox_target(anchors, labels):
    """Label anchor boxes using ground-truth bounding boxes."""
    batch_size, anchors                             = labels.shape[0], anchors.squeeze(0)
    batch_offset, batch_mask, batch_class_labels    = [], [], []

    device, num_anchors                             = anchors.device, anchors.shape[0]

    for i in range(batch_size):
        label               = labels[i, :, :]
        anchors_bbox_map    = assign_anchor_to_bbox(
                                label[:, 1:], anchors, device
                            )
        
        bbox_mask           = ((anchors_bbox_map >= 0).float().unsqueeze(-1)).repeat(1, 4)

        # Initialize class labels and assigned bounding box coordinates with
        # zeros
        class_labels        = torch.zeros(
                                num_anchors, 
                                dtype   = torch.long,
                                device  = device
                            )
        
        assigned_bb         = torch.zeros(
                                (num_anchors, 4), 
                                dtype   = torch.float32,
                                device  = device
                            )
        
        # Label classes of anchor boxes using their assigned ground-truth
        # bounding boxes. If an anchor box is not assigned any, we label its
        # class as background (the value remains zero)
        indices_true                = torch.nonzero(anchors_bbox_map >= 0)
        bb_idx                      = anchors_bbox_map[indices_true]
        class_labels[indices_true]  = label[bb_idx, 0].long() + 1
        assigned_bb[indices_true]   = label[bb_idx, 1:]

        # Offset transformation
        offset                      = offset_boxes(anchors, assigned_bb) * bbox_mask
        batch_offset.append(offset.reshape(-1))
        batch_mask.append(bbox_mask.reshape(-1))
        batch_class_labels.append(class_labels)

    bbox_offset     = torch.stack(batch_offset)
    bbox_mask       = torch.stack(batch_mask)
    class_labels    = torch.stack(batch_class_labels)
    
    return (bbox_offset, bbox_mask, class_labels)

1.4.4 Predicción de Cuadros Delimitadores con Supresión No Máxima

Durante la predicción, generamos múltiples cajas ancla para la imagen y predecimos las clases y desplazamientos para cada uno de ellos. Un cuadro delimitador predicho es por lo tanto, obtenido de acuerdo a la caja ancla con su desplazamiento predicho. A continuación, la función offset_inverse toma cajas ancla y predicciones de desplazamientos como entradas y aplica transformaciones para retornar las coordenadas del cuadro delimitador predicho.

Código Python · celda 124
def offset_inverse(anchors, offset_preds):
    """Predict bounding boxes based on anchor boxes with predicted offsets."""

    anc             = box_corner_to_center(anchors)
    pred_bbox_xy    = (offset_preds[:, :2] * anc[:, 2:] / 10) + anc[:, :2]
    pred_bbox_wh    = torch.exp(offset_preds[:, 2:] / 5) * anc[:, 2:]
    pred_bbox       = torch.cat((pred_bbox_xy, pred_bbox_wh), axis=1)
    predicted_bbox  = box_center_to_corner(pred_bbox)
    
    return predicted_bbox

Cuando hay muchas cajas ancla, muchos cuadros delimitadores predichos similares (con una superposición significante) pueden generarse potencialmente para rodear el mismo objeto. Para simplicar la salida, podemos mezclar los cuadros delimitadores predichos similares que pertenecen al mismo objeto usando la supresión no máxima (non-maximum suppression - NMS).

1.4.4.1 Funcionamiento de NMS

Para una predicción de un cuadro delimitador B, el modelo de detección de objetos calcula la probabilidad predicha para cada clase. Sea p la probabilidad predicha más grande, la clase correspondiente para esta probabilidad es la clase predicha para B. Específicamente, nos referimos a p como la confianza (puntuación o score) del cuadro delimitador para B. Sobre la misma imagen, todos los cuadros delimitadores predichos que no son de fondo se ordean por la confianza en orden descendente para generar una lista L. Luego manipulamos la lista ordenada L en los siguientes pasos:

  1. Seleccionar el cuadro delimitador predicho B1 con la confianza más alta de L como base y eliminar todos los cuadros delimitadores que no sean base cuyo IoU con B1 excede un umbral predefinido ϵ de L. En este punto, L mantiene el cuadro delimitador con la mayor confianza pero elimina los otros que son demasiado similares. En pocas palabras, se eliminan aquellos que puntuaciones de confianza no máximas.

  2. Seleccionar el cuadro delimitador predicho B2 con la segunda confianza más alta de L como otra base y eliminar todos los cuadros delimitadores que no sean base y cuyo IoU con B2 execede ϵ de L.

  3. Repetir el proceso (paso 1 al paso 2) hasta que todos los cuadros delimitadores de L han sido usados como una base. En este momento, la IoU de cualquier par de cuadros delimitadores predicho en L está poder debajo de un umbral ϵ; por lo tanto, ningún par es demasiado similar entre sí.

  4. La salida son todos los cuadros delimitadores predichos en la lista L.

La siguiente función nms ordena las puntuaciones de confianza en orden descendente y retorna sus índices.

Código Python · celda 127
def nms(boxes, scores, iou_threshold):
    """Sort confidence scores of predicted bounding boxes."""
    B           = torch.argsort(scores, dim=-1, descending=True)
    keep        = []  # Indices of predicted bounding boxes that will be kept

    while B.numel() > 0:

        i       = B[0]

        keep.append(i)
        if B.numel() == 1: break
        iou     = box_iou(
                    boxes[i, :].reshape(-1, 4),
                    boxes[B[1:], :].reshape(-1, 4)).reshape(-1)

        inds    = torch.nonzero(iou <= iou_threshold).reshape(-1)
        B       = B[inds + 1]
        
    return torch.tensor(keep, device=boxes.device)

Código Python · celda 129
def multibox_detection(
    cls_probs, 
    offset_preds, 
    anchors, 
    nms_threshold   = 0.5,
    pos_threshold   = 0.009999999
):
    """Predict bounding boxes using non-maximum suppression."""

    device, batch_size          = cls_probs.device, cls_probs.shape[0]
    anchors                     = anchors.squeeze(0)
    num_classes, num_anchors    = cls_probs.shape[1], cls_probs.shape[2]
    out                         = []

    for i in range(batch_size):

        cls_prob, offset_pred   = cls_probs[i], offset_preds[i].reshape(-1, 4)
        conf, class_id          = torch.max(cls_prob[1:], 0)

        predicted_bb            = offset_inverse(anchors, offset_pred)
        keep                    = nms(predicted_bb, conf, nms_threshold)

        # Find all non-`keep` indices and set the class to background
        all_idx                 = torch.arange(num_anchors, dtype=torch.long, device=device)
        combined                = torch.cat((keep, all_idx))
        uniques, counts         = combined.unique(return_counts=True)
        non_keep                = uniques[counts == 1]

        all_id_sorted           = torch.cat((keep, non_keep))
        class_id[non_keep]      = -1
        class_id                = class_id[all_id_sorted]
        conf, predicted_bb      = conf[all_id_sorted], predicted_bb[all_id_sorted]

        # Here `pos_threshold` is a threshold for positive (non-background)
        # predictions
        below_min_idx           = (conf < pos_threshold)
        class_id[below_min_idx] = -1
        conf[below_min_idx]     = 1 - conf[below_min_idx]
        pred_info               = torch.cat(
                                    (
                                        class_id.unsqueeze(1),
                                        conf.unsqueeze(1),
                                        predicted_bb
                                    ), 
                                    dim=1
                                )
        out.append(pred_info)
        
    return torch.stack(out)

1.5 Detección de Objetos Multiescala

En la sección anterior, se generaron múltiples cajas ancla centrada sobre cada píxel de una imagen de entrada. En escencia, estas cajas ancla representan muestras de diferentes regiones de la imagen. Sin embargo, podemos terminar con demasiadas cajas ancla para calcular si se generan para cada píxel. Por ej., si tenemos una imagen de entrada de 561×728; si se generar cinco cajas ancla con diferentes dimensiones para cada píxel como su centro, es necesario etiquetar y predecir en la imagen más de dos millones de cajas ancla, es decir, (561×728×5).

1.5.1 Cajas Ancla Multiescala

Para reducir el número de cajas ancla, por ejemplo, podemos muestrear uniformemente una pequeña porción de píxeles de la imagen de entrada para generar cajas ancla centradas en ellos. Además, a diferentes escalas podemos generar diferentes números de cajas ancla de diferentes tamaños. Intuitivamente, los objetos pequeños son más probables de aparecer en una imagen que objetos más grandes. Por ej., los objetos de 1×1, 1×2 y 2×2 pueden aparecer en una imagen de 2×2 de 4,2 y 1 formas posibles, respectivamente. Por lo tanto, cuando usamos cajas ancla más pequeñas para detectar objetos más pequeños, podemos muestrar más regiones, mientras que para objetos más grandes podemos muestrear menos regiones.

def multibox_prior(data, sizes, ratios):
    """Generate anchor boxes with different shapes centered on each pixel."""
    
    print(f"data.shape: {data.shape}")

    in_height, in_width             = data.shape[-2:]
    device, num_sizes, num_ratios   = data.device, len(sizes), len(ratios)
    boxes_per_pixel                 = (num_sizes + num_ratios - 1)
    size_tensor                     = torch.tensor(sizes, device=device)
    ratio_tensor                    = torch.tensor(ratios, device=device)

    # Offsets are required to move the anchor to the center of a pixel. Since
    # a pixel has height=1 and width=1, we choose to offset our centers by 0.5
    offset_h, offset_w              = 0.5, 0.5
    steps_h                         = 1.0 / in_height  # Scaled steps in y axis
    steps_w                         = 1.0 / in_width  # Scaled steps in x axis

    # Generate all center points for the anchor boxes
    center_h                        = (torch.arange(in_height, device=device) + offset_h) * steps_h
    center_w                        = (torch.arange(in_width, device=device) + offset_w) * steps_w
    shift_y, shift_x                = torch.meshgrid(center_h, center_w)
    shift_y, shift_x                = shift_y.reshape(-1), shift_x.reshape(-1) # size = in_height * in_width
    
    print(f"center_h: {center_h.shape} \n center_w: {center_w.shape} \n")
    print(f"shift_y: {shift_y.shape} \n shift_x: {shift_x.shape} \n")

    # Generate `boxes_per_pixel` number of heights and widths that are later
    # used to create anchor box corner coordinates (xmin, xmax, ymin, ymax)
    w = torch.cat((size_tensor * torch.sqrt(ratio_tensor[0]),
                   sizes[0] * torch.sqrt(ratio_tensor[1:])))\
                   * in_height / in_width  # Handle rectangular inputs

    h = torch.cat(
        (
            size_tensor / torch.sqrt(ratio_tensor[0]),
            sizes[0] / torch.sqrt(ratio_tensor[1:])
        )
    )

    print(f"w: {w.shape} \n h: {h.shape} \n")


    # Divide by 2 to get half height and half width
    anchor_manipulations = torch.stack(
        (-w, -h, w, h)
    ).T.repeat(in_height * in_width, 1) / 2

    print(f"anchor_manipulations: {anchor_manipulations.shape} \n")

    # Each center point will have `boxes_per_pixel` number of anchor boxes, so
    # generate a grid of all anchor box centers with `boxes_per_pixel` repeats
    out_grid = torch.stack(
        [
            shift_x, 
            shift_y, 
            shift_x, 
            shift_y
        ],
        dim=1).repeat_interleave(boxes_per_pixel, dim=0)

    output = out_grid + anchor_manipulations

    return output.unsqueeze(0)

def bbox_to_rect(bbox, color):
    """Convert bounding box to matplotlib format."""
    # Convert the bounding box (upper-left x, upper-left y, lower-right x,
    # lower-right y) format to the matplotlib format: ((upper-left x,
    # upper-left y), width, height)
    return plt.Rectangle(
        xy          = (bbox[0], bbox[1]), 
        width       = bbox[2]-bbox[0], 
        height      = bbox[3]-bbox[1],
        fill        = False, 
        edgecolor   = color, 
        linewidth   = 2
    )

def show_bboxes(axes, bboxes, labels=None, colors=None):
    """Show bounding boxes."""

    def make_list(obj, default_values=None):
        if obj is None:
            obj = default_values
        elif not isinstance(obj, (list, tuple)):
            obj = [obj]
        return obj

    labels      = make_list(labels)
    colors      = make_list(colors, ['b', 'g', 'r', 'm', 'c'])

    for i, bbox in enumerate(bboxes):
        color   = colors[i % len(colors)]
        rect    = bbox_to_rect(bbox.detach().numpy(), color)

        axes.add_patch(rect)
        if labels and len(labels) > i:
            text_color  = 'k' if color == 'w' else 'w'
            axes.text(
                rect.xy[0], 
                rect.xy[1], 
                labels[i],
                va          = 'center', 
                ha          = 'center', 
                fontsize    = 9, 
                color       = text_color,
                bbox        = dict(facecolor=color, lw=0)
            )
